"""Collect actual initial/revised LLM responses with Ollama; resumable JSONL."""
from __future__ import annotations

import json
import math
import re
import time
from decimal import Decimal, InvalidOperation
from pathlib import Path
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen

from .data import collection_protocol, question_hash, transition_label

INITIAL_INSTRUCTION = "Solve the question. Give a brief explanation, then finish with FINAL: followed by your answer."
RECHECK_INSTRUCTION = ("Recheck your previous answer against the question. Keep it if justified; change it only if you identify an error. "
                       "Give a brief explanation, then finish with FINAL: followed by your answer.")


class OllamaResponseError(ValueError):
    """An invalid returned call, retained only as unvalidated failure evidence."""

    def __init__(self, message, response):
        super().__init__(message)
        self.response = response


def parse_answer(text, answer_type):
    # Require an explicit answer marker; do not guess the last arithmetic intermediate.
    marked=re.findall(r"(?:FINAL(?:\s+ANSWER)?|Answer)\s*:\s*([^\n]+)",str(text),re.I)
    candidate=marked[-1].strip() if marked else str(text).strip()
    candidate=candidate.strip(" *`.$")
    if answer_type=="boolean":
        value=candidate.lower().strip(" .!")
        return "yes" if value in ("yes","true") else "no" if value in ("no","false") else None
    if answer_type=="choice":
        value=candidate.upper().strip(" .)")
        return value if len(value)==1 and value in "ABCDE" else None
    if answer_type=="number":
        candidate=candidate.replace(",","").strip().rstrip(".")
        if not re.fullmatch(r"[-+]?\d+(?:\.\d+)?",candidate):
            return None
        try:
            return str(Decimal(candidate).normalize())
        except InvalidOperation:
            return None
    raise ValueError("answer_type must be number, boolean, or choice")


def ollama_chat(model, messages, host, budget, seed, temperature, thinking, *, timeout=600, context_window=None, decoding_options=None):
    payload=dict(model=model,messages=messages,stream=False,think=thinking,
                 options=dict(num_predict=budget,seed=seed,temperature=temperature))
    if context_window is not None:
        if isinstance(context_window, bool) or not isinstance(context_window, int) or context_window < 1:
            raise ValueError("context_window must be a positive integer when supplied.")
        payload["options"]["num_ctx"] = context_window
    if decoding_options is not None:
        if not isinstance(decoding_options, dict) or set(decoding_options) - {"top_k", "top_p", "min_p", "repeat_penalty", "stop"}:
            raise ValueError("Unsupported explicit decoding options.")
        payload["options"].update(decoding_options)
    request=Request(host.rstrip("/")+"/api/chat",data=json.dumps(payload).encode(),
                    headers={"Content-Type":"application/json"})
    started=time.perf_counter()
    try:
        with urlopen(request,timeout=timeout) as response:
            result=json.load(response)
    except HTTPError as exc:
        # Ollama includes the runner failure in its JSON body. Read a bounded
        # amount so a generic HTTP 500 does not hide memory/context diagnostics.
        body = exc.read(4097)
        truncated = len(body) > 4096
        text = body[:4096].decode("utf-8", errors="replace").strip()
        try:
            payload = json.loads(text)
            detail = payload.get("error") if isinstance(payload, dict) else text
            detail = detail if isinstance(detail, str) and detail.strip() else text
        except (ValueError, TypeError):
            detail = text
        detail = detail or str(exc.reason)
        if truncated:
            detail += " [server error body truncated]"
        raise URLError(f"Ollama HTTP {exc.code}: {detail}") from exc
    if not isinstance(result, dict) or result.get("done") is not True or "message" not in result:
        raise OllamaResponseError("Ollama returned an incomplete or invalid response.", result)
    result["wall_seconds"]=time.perf_counter()-started
    return result


def response_text(result):
    message=result["message"]
    thinking=message.get("thinking","")
    content=message.get("content","")
    return (thinking+"\n\n"+content).strip() if thinking else content


def actual_tokens(result):
    if "prompt_eval_count" not in result or "eval_count" not in result:
        raise ValueError("Ollama did not provide actual token counts. Refusing to estimate them.")
    counts = []
    for key in ("prompt_eval_count", "eval_count"):
        value = result[key]
        if isinstance(value, bool):
            raise ValueError(f"Ollama {key} must be a finite nonnegative integer.")
        try:
            count = float(value)
        except (TypeError, ValueError, OverflowError) as exc:
            raise ValueError(f"Ollama {key} must be a finite nonnegative integer.") from exc
        if not math.isfinite(count) or count < 0 or not count.is_integer():
            raise ValueError(f"Ollama {key} must be a finite nonnegative integer.")
        counts.append(int(count))
    return sum(counts)


def collect_pairs(input_path, output_path, model="qwen3:1.7b", host="http://localhost:11434", budget=768,
                  seed=42, temperature=0.0, thinking=False, limit=None, *, context_window=None):
    output=Path(output_path)
    output.parent.mkdir(parents=True,exist_ok=True)
    metadata_path=output.with_suffix(".metadata.json")
    config=dict(model=model,host=host,budget=budget,seed=seed,temperature=temperature,thinking=thinking,
                initial_instruction=INITIAL_INSTRUCTION,recheck_instruction=RECHECK_INSTRUCTION,protocol_version="0.1")
    if context_window is not None:
        config.update(context_window=context_window, protocol_version="0.2")
    collection_protocol(config)  # Validate before writing metadata or making calls.
    chat_options = {} if context_window is None else {"context_window": context_window}
    saved_config=json.loads(metadata_path.read_text(encoding="utf-8")) if metadata_path.exists() else None
    saved_parameters={k:v for k,v in saved_config.items() if k!="model_digest"} if saved_config else None
    if saved_parameters is not None and saved_parameters!=config:
        raise ValueError("Generation settings differ from this saved run. Use a new output filename.")
    existing={}
    if output.exists():
        for line in output.read_text(encoding="utf-8").splitlines():
            if line.strip():
                row=json.loads(line)
                existing[str(row["example_id"])]=row
        if not metadata_path.exists():
            raise ValueError("Existing output has no protocol metadata; use a new output filename.")
    questions=[json.loads(line) for line in Path(input_path).read_text(encoding="utf-8").splitlines() if line.strip()]
    if len({str(q["example_id"]) for q in questions})!=len(questions):
        raise ValueError("Input contains duplicate example IDs.")
    if limit:
        questions=questions[:limit]
    for question in questions:
        if parse_answer(question["gold_answer"],question["answer_type"]) is None:
            raise ValueError(f"Cannot normalize the answer key for {question['example_id']}")
        if str(question["example_id"]) in existing:
            if question_hash(question["query"])!=question_hash(existing[str(question["example_id"])]["query"]):
                raise ValueError("A resumed example ID now refers to a different question.")
    # One preflight avoids repeated failed inference calls when the server/model is unavailable.
    try:
        with urlopen(host.rstrip("/")+"/api/tags",timeout=10) as response:
            installed={m["name"]:m.get("digest","") for m in json.load(response).get("models",[])}
    except (URLError,TimeoutError) as exc:
        raise RuntimeError("Local Ollama is unavailable. Install/start Ollama before collecting new responses.") from exc
    if model not in installed:
        raise RuntimeError(f"Model {model} is not installed. Run: ollama pull {model}")
    config["model_digest"]=installed[model]
    if saved_config is not None and saved_config.get("model_digest")!=config["model_digest"]:
        raise ValueError("The installed model digest changed. Use a new output filename to keep runs separate.")
    metadata_path.write_text(json.dumps(config,indent=2)+"\n",encoding="utf-8")
    completed=0
    for index, question in enumerate(questions,1):
        if str(question["example_id"]) in existing:
            print(f"[{index}/{len(questions)}] already saved",flush=True)
            continue
        user=INITIAL_INSTRUCTION+"\n\n"+question["query"]
        messages=[dict(role="user",content=user)]
        # Neither generation call receives the benchmark answer key.
        base=ollama_chat(model,messages,host,budget,seed,temperature,thinking,**chat_options)
        previous=base["message"].get("content","") or response_text(base)
        revised=ollama_chat(model,messages+[dict(role="assistant",content=previous),
            dict(role="user",content=RECHECK_INSTRUCTION)],host,budget,seed+1,temperature,thinking,**chat_options)
        base_answer=parse_answer(base["message"].get("content",""),question["answer_type"])
        final_answer=parse_answer(revised["message"].get("content",""),question["answer_type"])
        gold=parse_answer(question["gold_answer"],question["answer_type"])
        if gold is None:
            raise ValueError(f"Cannot normalize the answer key for {question['example_id']}")
        base_correct=base_answer is not None and base_answer==gold
        final_correct=final_answer is not None and final_answer==gold
        row=dict(example_id=question["example_id"],dataset=question["dataset"],query=question["query"],
            gold_answer=gold,answer_type=question["answer_type"],model=model,rollout=0,seed=seed,
            base_response=response_text(base),base_answer=base_answer or "",base_correct=base_correct,
            action="recheck",intervention_response=response_text(revised),final_answer=final_answer or "",
            final_correct=final_correct,label=transition_label(base_correct,final_correct),
            base_actual_tokens=actual_tokens(base),action_actual_tokens=actual_tokens(revised),
            base_done_reason=base.get("done_reason",""),intervention_done_reason=revised.get("done_reason",""),
            base_finalizer_used=False,base_parse_valid=base_answer is not None,final_parse_valid=final_answer is not None,
            base_wall_seconds=base["wall_seconds"],action_wall_seconds=revised["wall_seconds"],
            model_digest=config["model_digest"],
            provenance="New local Ollama generation; fixed initial answer followed by one explicit self-recheck")
        with output.open("a",encoding="utf-8") as handle:
            handle.write(json.dumps(row,ensure_ascii=False)+"\n")
        completed+=1
        print(f"[{index}/{len(questions)}] {row['label']} | {row['base_actual_tokens']} + {row['action_actual_tokens']} tokens",flush=True)
    return completed

"""Validate paired outcomes and keep every rollout of a question together."""
from __future__ import annotations

import hashlib
import json
import unicodedata
from pathlib import Path

import numpy as np
import pandas as pd

from .features import INITIAL_REQUIRED, validate_initial_record

LABELS = ["HELP", "NO_CHANGE", "HURT"]
REQUIRED = [*INITIAL_REQUIRED, "base_correct", "final_correct", "action_actual_tokens",
            "action", "model", "recheck_protocol"]
PUBLIC_REVISION = "10411137fcaa02b3bd549a2285b43ffe315417f6"
COLLECTION_SETTINGS = ("protocol_version", "budget", "seed", "temperature", "thinking",
                       "initial_instruction", "recheck_instruction")


def as_bool(value):
    if isinstance(value, (bool, np.bool_)):
        return bool(value)
    if isinstance(value, (int, float, np.integer, np.floating)) and value in (0, 1):
        return bool(value)
    if str(value).strip().lower() in ("true", "1"):
        return True
    if str(value).strip().lower() in ("false", "0"):
        return False
    raise ValueError(f"Expected a boolean correctness label, got {value!r}")


def transition_label(base_correct, final_correct):
    base, final = as_bool(base_correct), as_bool(final_correct)
    return "HELP" if not base and final else "HURT" if base and not final else "NO_CHANGE"


def question_hash(query):
    if not isinstance(query, str) or not query.strip():
        raise ValueError("Question text must be a nonempty string.")
    canonical = " ".join(unicodedata.normalize("NFKC", query).casefold().split())
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


def _require_text_column(frame, key):
    if key not in frame:
        raise ValueError(f"{key} is required to identify the fixed model/recheck protocol.")
    values = frame[key]
    if not values.map(lambda value: isinstance(value, str) and bool(value.strip())).all():
        raise ValueError(f"{key} must contain explicit nonempty strings.")


def _single_value(frame, key):
    _require_text_column(frame, key)
    values = frame[key].drop_duplicates().tolist()
    if len(values) != 1:
        raise ValueError(f"Use one fixed {key} per experiment; found {len(values)} values.")
    return values[0]


def _token_counts(values, key):
    if values.map(lambda value: isinstance(value, (bool, np.bool_))).any():
        raise ValueError(f"Invalid token counts in {key}: booleans are not token counts.")
    try:
        counts = pd.to_numeric(values, errors="raise")
    except (ValueError, TypeError) as exc:
        raise ValueError(f"Invalid token counts in {key}.") from exc
    if not np.isfinite(counts).all() or (counts < 0).any() or (counts % 1 != 0).any():
        raise ValueError(f"Invalid token counts in {key}: require finite nonnegative integers.")
    return counts


def validate_pairs(frame: pd.DataFrame) -> pd.DataFrame:
    if not isinstance(frame, pd.DataFrame) or frame.empty:
        raise ValueError("Paired outcomes must be a nonempty DataFrame.")
    missing = set(REQUIRED) - set(frame.columns)
    if missing:
        raise ValueError(f"Missing columns: {', '.join(sorted(missing))}")
    out = frame.copy()
    # This is the same initial-input validation used by training and the demo.
    initial = [validate_initial_record(row) for row in out.to_dict("records")]
    for key in INITIAL_REQUIRED:
        out[key] = [row[key] for row in initial]
    for key in ("base_correct", "final_correct"):
        out[key] = out[key].map(as_bool)
    out["label"] = [transition_label(a, b) for a, b in zip(out.base_correct, out.final_correct)]
    out["group_id"] = out["query"].map(question_hash)
    for key in ("base_actual_tokens", "action_actual_tokens"):
        out[key] = _token_counts(out[key], key)
    for key in ("action", "model", "recheck_protocol"):
        _single_value(out, key)
    if out["action"].iloc[0] not in ("recheck", "active_verify"):
        raise ValueError("Only one explicit recheck/active_verify action belongs in a paired rechecking experiment.")
    if "model_digest" in out:
        _single_value(out, "model_digest")
    if "rollout" not in out:
        out["rollout"] = 0
    out["rollout"] = _token_counts(out["rollout"], "rollout")
    if "example_id" not in out:
        out["example_id"] = out["group_id"].str[:16]
    if "dataset" not in out:
        out["dataset"] = "user_pairs"
    for key in ("example_id", "dataset"):
        if out[key].isna().any() or out[key].astype(str).str.strip().eq("").any():
            raise ValueError(f"{key} cannot be missing or empty.")
        out[key] = out[key].astype(str)
    # All observations sharing an attempt must describe one fixed initial call.
    out["attempt_id"] = out.dataset.astype(str) + ":" + out.example_id.astype(str)
    initial_keys = ["group_id", *INITIAL_REQUIRED, "base_correct"]
    # Question case/spacing is canonicalized; initial response metadata stays exact.
    initial_keys.remove("query")
    consistency = out.groupby(["dataset", "example_id"])[initial_keys].nunique(dropna=False)
    if (consistency > 1).any().any():
        raise ValueError("A repeated attempt has inconsistent initial responses or metadata. Fix its rollout IDs.")
    return out.reset_index(drop=True)


def collection_protocol(metadata):
    """Identify actual generation settings; legacy hashes omit unspecified context."""
    missing = set(COLLECTION_SETTINGS) - set(metadata)
    if missing:
        raise ValueError("Collection metadata is missing generation settings: " + ", ".join(sorted(missing)))
    settings = {key: metadata[key] for key in COLLECTION_SETTINGS}
    for key in ("protocol_version", "initial_instruction", "recheck_instruction"):
        if not isinstance(settings[key], str) or not settings[key].strip():
            raise ValueError(f"Collection metadata requires a nonempty {key}.")
    for key in ("budget", "seed"):
        if isinstance(settings[key], bool) or not isinstance(settings[key], int):
            raise ValueError(f"Collection metadata {key} must be an integer.")
    if settings["budget"] < 1 or not isinstance(settings["thinking"], bool):
        raise ValueError("Collection metadata needs a positive budget and explicit thinking boolean.")
    try:
        temperature = float(settings["temperature"])
    except (ValueError, TypeError, OverflowError) as exc:
        raise ValueError("Collection temperature must be finite and nonnegative.") from exc
    if isinstance(settings["temperature"], bool) or not np.isfinite(temperature) or temperature < 0:
        raise ValueError("Collection temperature must be finite and nonnegative.")
    settings["temperature"] = temperature
    if "context_window" in metadata:
        context_window = metadata["context_window"]
        if isinstance(context_window, bool) or not isinstance(context_window, int) or context_window < 1:
            raise ValueError("Collection metadata context_window must be a positive integer.")
        settings["context_window"] = context_window
    serialized = json.dumps(settings, sort_keys=True, separators=(",", ":"), allow_nan=False)
    return "ollama:" + hashlib.sha256(serialized.encode("utf-8")).hexdigest(), settings


def _attach_collection_metadata(frame, path):
    metadata_path = path.with_suffix(".metadata.json")
    if not metadata_path.exists():
        return frame
    metadata = json.loads(metadata_path.read_text(encoding="utf-8"))
    if not isinstance(metadata, dict):
        raise ValueError("Collection protocol metadata must be an object.")
    protocol, settings = collection_protocol(metadata)
    out = frame.copy()
    identities = {"model": metadata.get("model"), "recheck_protocol": protocol}
    if "model_digest" in metadata:
        identities["model_digest"] = metadata["model_digest"]
    for key, value in identities.items():
        if not isinstance(value, str) or not value.strip():
            raise ValueError(f"Collection metadata requires an explicit {key}.")
        if key in out and not out[key].eq(value).all():
            raise ValueError(f"Row {key} conflicts with its collection protocol metadata.")
        out[key] = value
    out.attrs["collection_metadata"] = metadata
    out.attrs["generation_settings"] = settings
    return out


def load_pairs(path) -> pd.DataFrame:
    path = Path(path)
    if path.suffix.lower() == ".parquet":
        frame = pd.read_parquet(path)
    elif path.suffix.lower() == ".jsonl":
        frame = pd.read_json(path, lines=True, dtype=False)
    else:
        # Answer/ID strings may contain significant leading zeroes. Numeric
        # token fields are validated separately after loading.
        frame = pd.read_csv(path, keep_default_na=False,
                            dtype={key: str for key in ("base_answer", "final_answer", "gold_answer", "example_id")})
    return validate_pairs(_attach_collection_metadata(frame, path))


def validate_transfer_compatibility(source, target):
    """Reject solver/protocol changes before fitting a transferred gate."""
    assert_disjoint(source, target)
    identity = {}
    for key in ("model", "recheck_protocol", "action"):
        left, right = _single_value(source, key), _single_value(target, key)
        if left != right:
            raise ValueError(f"Source/target {key} compatibility mismatch: {left!r} != {right!r}.")
        identity[key] = left
    if "model_digest" in source or "model_digest" in target:
        left, right = _single_value(source, "model_digest"), _single_value(target, "model_digest")
        if left != right:
            raise ValueError("Source/target model checkpoint compatibility mismatch.")
        identity["model_digest"] = left
    if "action_description" in source or "action_description" in target:
        left, right = _single_value(source, "action_description"), _single_value(target, "action_description")
        if left != right:
            raise ValueError("Source/target recheck action descriptions differ.")
        identity["action_description_sha256"] = hashlib.sha256(left.encode("utf-8")).hexdigest()
    settings = source.attrs.get("generation_settings")
    target_settings = target.attrs.get("generation_settings")
    if settings is not None or target_settings is not None:
        if target_settings != settings:
            raise ValueError("Source/target generation settings differ.")
        identity["generation_settings"] = settings
    public_settings = source.attrs.get("public_generation_settings")
    target_public_settings = target.attrs.get("public_generation_settings")
    if public_settings is not None or target_public_settings is not None:
        if target_public_settings != public_settings:
            raise ValueError("Source/target public generation settings differ.")
        identity["published_generation_settings"] = public_settings
        identity["identity_validation_limit"] = (
            "Public release identifies model family and action, but omits the checkpoint digest and full prompts."
        )
    elif "model_digest" not in identity:
        identity["identity_validation_limit"] = "Model name and declared protocol match; no checkpoint digest was supplied."
    return identity


def _with_canonical_groups(frame):
    if "query" not in frame:
        raise ValueError("Question text is required for independent-question grouping.")
    out = frame.copy()
    out["group_id"] = out["query"].map(question_hash)
    return out


def group_partition(frame, fractions, seed=42):
    if not fractions or not np.isfinite(fractions).all() or not np.isclose(sum(fractions), 1) or any(f <= 0 for f in fractions):
        raise ValueError("Positive partition fractions must sum to one.")
    frame = _with_canonical_groups(frame)
    groups = frame.group_id.drop_duplicates().to_numpy()
    np.random.default_rng(seed).shuffle(groups)
    bounds = [0] + [int(len(groups) * sum(fractions[:i])) for i in range(1, len(fractions))] + [len(groups)]
    parts = [frame[frame.group_id.isin(groups[bounds[i]:bounds[i + 1]])].copy().reset_index(drop=True)
             for i in range(len(fractions))]
    if any(p.empty for p in parts):
        raise ValueError("Too few independent questions for the requested split.")
    assert_disjoint(*parts)
    return parts


def assert_disjoint(*frames):
    # Never trust cached/user-provided hashes when proving no question leakage.
    if any("query" not in f for f in frames):
        raise ValueError("Question text is required to check partition leakage.")
    groups = [set(f["query"].map(question_hash)) for f in frames]
    for i in range(len(groups)):
        for j in range(i + 1, len(groups)):
            if groups[i] & groups[j]:
                raise ValueError("Question leakage between experimental partitions.")


def public_frames(root: Path):
    raw = Path(root) / "data" / "raw"
    source = pd.read_parquet(raw / "train.parquet")
    target = pd.read_parquet(raw / "commonsenseqa.parquet")
    source = source[source.action == "active_verify"].copy()
    target = target[target.action == "active_verify"].copy()
    source["dataset"], target["dataset"] = "SEVRA_MATH_train", "SEVRA_CommonsenseQA"
    for frame in (source, target):
        if _single_value(frame, "model") != "qwen3:4b":
            raise ValueError("Included public data model differs from its pinned Qwen3-4B attribution.")
        frame["recheck_protocol"] = f"sevra:{PUBLIC_REVISION}:active_verify"
        # Published unavailable completion statuses become a training-time category.
        # Raw files remain untouched; the feature builder flags unknown completion.
        unknown = frame["base_done_reason"].eq("")
        frame.loc[unknown, "base_done_reason"] = "unknown"
        frame.attrs["unknown_initial_completion_rows"] = int(unknown.sum())
        frame.attrs["public_generation_settings"] = {
            "dataset_revision": PUBLIC_REVISION, "base_generation_limit": 4096,
            "finalizer_max_tokens": 512, "finalizer": "non-reasoning final answer when needed",
            "protocol_evidence": "Pinned public dataset card; complete prompts/checkpoint digest not released",
        }
    return validate_pairs(source), validate_pairs(target)


def write_manifest(path, partitions, metadata):
    payload = dict(metadata)
    payload["partitions"] = {
        name: {"questions": int(f.group_id.nunique()), "rows": len(f),
               "group_ids": sorted(f.group_id.unique().tolist()), "label_counts": f.label.value_counts().to_dict()}
        for name, f in partitions.items()
    }
    Path(path).write_text(json.dumps(payload, indent=2) + "\n", encoding="utf-8")

"""Frozen source model, reserved target threshold adaptation, fixed test labels."""
from __future__ import annotations

import hashlib
import importlib.metadata

import numpy as np
import pandas as pd

from .data import (assert_disjoint, group_partition, public_frames, validate_pairs,

def run_experiment(root, source=None, target=None, seed=42, repeats=5,
                   sizes=(25, 50, 100, 200, 400), harm_weight=1.0):
    if (source is None) != (target is None):
        raise ValueError("Provide both custom source and target datasets, or neither.")
    if repeats < 1 or int(repeats) != repeats or not np.isfinite(harm_weight) or harm_weight <= 0:
        raise ValueError("Repeats must be a positive integer; harm weight must be finite and positive.")
    sizes = tuple(sizes)
    if not sizes or any(not isinstance(n, (int, np.integer)) or n <= 0 for n in sizes) or len(set(sizes)) != len(sizes):
        raise ValueError("Adaptation sizes must be distinct positive integer question counts.")
    root = Path(root).resolve()
    if root == PROJECT_ROOT:
        raise ValueError("Choose a separate output folder; the shipped v0.1 baseline is preserved.")
    started = time.perf_counter()
    is_public = source is None
    if is_public:
        source, target = public_frames(PROJECT_ROOT)
        provenance = "Published Qwen3-4B SEVRA outcomes; secondary replay analysis, no new LLM inference."
    else:
        source, target = validate_pairs(source), validate_pairs(target)

===== features.py =====

"""The stable pre-recheck allowlist, shared by training, evaluation, and replay.

Every initial signal must be supplied. An explicitly empty parsed initial
answer means parsing failed. An explicitly 'unknown' completion reason is
represented by its own indicator and is used for missing public-log statuses.
Gold labels, revised text, and future token counts are never read here.
"""
from __future__ import annotations

import hashlib
import json
import math
import re

import numpy as np
import pandas as pd

UNCERTAIN = re.compile(r"\b(maybe|perhaps|unsure|uncertain|possibly|might|not sure|i think)\b", re.I)
BACKTRACK = re.compile(r"\b(wait|however|instead|mistake|reconsider|actually|wrong)\b", re.I)
CHECK = re.compile(r"\b(check|verify|verification|confirm|recheck)\b", re.I)
WORDS = re.compile(r"\b\w+\b")

INITIAL_REQUIRED = (
    "query", "base_response", "base_answer", "base_actual_tokens",
    "base_done_reason", "base_finalizer_used",
)
FEATURE_SCHEMA_VERSION = "pre-recheck-v2"
FEATURE_NAMES = [
    "question_log_chars", "question_log_words", "question_numbers",
    "question_math_symbols", "question_marks", "initial_log_chars",
    "initial_log_words", "initial_unique_word_ratio",
    "initial_uncertainty_per_100_words", "initial_backtracking_per_100_words",
    "initial_checking_per_100_words", "initial_number_count",
    "initial_paragraph_count", "initial_question_marks",
    "initial_log_actual_tokens", "initial_finalizer_used",
    "initial_budget_exhausted", "initial_answer_present", "initial_completion_unknown",
]
FEATURE_SCHEMA_FINGERPRINT = hashlib.sha256(json.dumps({
    "version": FEATURE_SCHEMA_VERSION, "features": FEATURE_NAMES,
    "required_inputs": INITIAL_REQUIRED, "missing_data": "reject; explicit unknown completion indicator",
    "completion_values": ["stop", "length", "unknown"], "initial_tokens": "finite nonnegative integer",
}, sort_keys=True, separators=(",", ":")).encode("utf-8")).hexdigest()


def validate_initial_record(row):
    """Validate only the inputs available before a recheck and normalize types."""
    missing = [key for key in INITIAL_REQUIRED if key not in row]
    if missing:
        raise ValueError("Missing required initial signals: " + ", ".join(missing))
    out = {}
    for key in ("query", "base_response", "base_answer", "base_done_reason"):
        value = row[key]
        if value is None or not np.isscalar(value) or pd.isna(value):
            raise ValueError(f"{key} must be supplied; use an explicit empty base_answer for parsing failure.")
        if key != "base_answer" and not isinstance(value, str):
            raise ValueError(f"{key} must be text.")
        out[key] = str(value)
    if not out["query"].strip():
        raise ValueError("query cannot be empty.")
    if not out["base_done_reason"].strip():
        raise ValueError("base_done_reason cannot be empty; supply the recorded completion status or explicit 'unknown'.")
    out["base_done_reason"] = out["base_done_reason"].strip().casefold()
    if out["base_done_reason"] not in ("stop", "length", "unknown"):
        raise ValueError("base_done_reason must be stop, length, or explicit unknown.")
    value = row["base_actual_tokens"]
    if isinstance(value, (bool, np.bool_)):
        raise ValueError("base_actual_tokens must be a finite nonnegative number.")
    try:
        tokens = float(value)
    except (ValueError, TypeError, OverflowError) as exc:
        raise ValueError("base_actual_tokens must be a finite nonnegative number.") from exc
    if not math.isfinite(tokens) or tokens < 0 or not tokens.is_integer():
        raise ValueError("base_actual_tokens must be a finite nonnegative integer.")
    out["base_actual_tokens"] = tokens
    value = row["base_finalizer_used"]
    if isinstance(value, (bool, np.bool_)):
        finalizer = bool(value)
    elif isinstance(value, (int, float, np.integer, np.floating)) and value in (0, 1):
        finalizer = bool(value)
    elif isinstance(value, str) and value.strip().lower() in ("true", "false", "1", "0"):
        finalizer = value.strip().lower() in ("true", "1")
    else:
        raise ValueError("base_finalizer_used must be an explicit boolean or 0/1.")
    out["base_finalizer_used"] = finalizer
    return out


def extract_one(row):
    row = validate_initial_record(row)
    q, a = row["query"], row["base_response"]
    qw, aw = WORDS.findall(q.lower()), WORDS.findall(a.lower())
    n = max(len(aw), 1)
    values = {
        "question_log_chars": math.log1p(len(q)),
        "question_log_words": math.log1p(len(qw)),
        "question_numbers": len(re.findall(r"\d+(?:\.\d+)?", q)),
        "question_math_symbols": sum(q.count(c) for c in "+-=*/^\\"),
        "question_marks": q.count("?"),
        "initial_log_chars": math.log1p(len(a)),
        "initial_log_words": math.log1p(len(aw)),
        "initial_unique_word_ratio": len(set(aw)) / n,
        "initial_uncertainty_per_100_words": 100 * len(UNCERTAIN.findall(a)) / n,
        "initial_backtracking_per_100_words": 100 * len(BACKTRACK.findall(a)) / n,
        "initial_checking_per_100_words": 100 * len(CHECK.findall(a)) / n,
        "initial_number_count": len(re.findall(r"\d+(?:\.\d+)?", a)),
        "initial_paragraph_count": a.count("\n\n") + 1,
        "initial_question_marks": a.count("?"),
        "initial_log_actual_tokens": math.log1p(row["base_actual_tokens"]),
        "initial_finalizer_used": int(row["base_finalizer_used"]),
        "initial_budget_exhausted": int(row["base_done_reason"] == "length"),
        "initial_answer_present": int(bool(row["base_answer"].strip())),
        "initial_completion_unknown": int(row["base_done_reason"] == "unknown"),
    }
    return {key: values[key] for key in FEATURE_NAMES}


def extract_features(frame):
    if not isinstance(frame, pd.DataFrame):
        raise ValueError("Initial inputs must be supplied as a pandas DataFrame.")
    missing = [key for key in INITIAL_REQUIRED if key not in frame.columns]
    if missing:
        raise ValueError("Missing required initial signals: " + ", ".join(missing))
    records = []
    for index, row in zip(frame.index, frame.to_dict("records")):
        try:
            records.append(extract_one(row))
        except ValueError as exc:
            raise ValueError(f"Invalid initial inputs at row {index}: {exc}") from exc
    return pd.DataFrame(records, columns=FEATURE_NAMES, index=frame.index, dtype=float)


===== live.py =====

"""Question-only live Ollama execution: an initial call followed by zero or one recheck.

The public gate can be demonstrated with its named solver under an explicitly
experimental collector protocol. Its released prompts and checkpoint digest
are unavailable, so this live use is not the evaluated replay protocol.
Custom gates require a complete, matching collector protocol and checkpoint.
There are no answer keys or correctness-based overrides in this engine.
"""
from __future__ import annotations

import hashlib
import importlib
import json
import math
from pathlib import Path
import re
import time
from urllib.error import URLError
from urllib.parse import urlparse
from urllib.request import urlopen

import pandas as pd

from .collector import (
    INITIAL_INSTRUCTION, RECHECK_INSTRUCTION, actual_tokens,
    ollama_chat, parse_answer, response_text,
)
from .data import COLLECTION_SETTINGS, collection_protocol
from .features import INITIAL_REQUIRED, validate_initial_record
from .predictor import ArtifactLoadError, decide_recheck, load_artifact

DEFAULT_HOST = "http://localhost:11434"
DEFAULT_TIMEOUT = 120
DEFAULT_ARTIFACT = Path(__file__).resolve().parents[1] / "outputs" / "v0.2" / "models" / "source_gate.joblib"
DEFAULT_SETTINGS = {
    "protocol_version": "0.2", "budget": 768, "seed": 42, "context_window": 4096,
    "temperature": 0.0, "thinking": False,
    "initial_instruction": INITIAL_INSTRUCTION,
    "recheck_instruction": RECHECK_INSTRUCTION,
}


def load_live_artifact(path=DEFAULT_ARTIFACT, *, expected_protocol=None, expected_sha256=None):
    """Freshly deserialize the production bundle; never cache a class instance.

    Streamlit can evict imported modules while retaining resource-cache objects.
    Loading from disk resolves OutcomePredictor through its current canonical
    import, preserving strict isinstance and all fitted/schema checks.
    """
    return load_artifact(path, expected_protocol=expected_protocol, expected_sha256=expected_sha256)


def protocol_fingerprint(host, model, model_digest, settings):
    """Scope a UI run identity to the actual endpoint, solver, and call settings."""
    payload = dict(host=host.rstrip("/"), model=model, model_digest=model_digest,
                   generation_settings=settings)
    return hashlib.sha256(json.dumps(payload, sort_keys=True, separators=(",", ":"),
                                     allow_nan=False).encode("utf-8")).hexdigest()


def _host(host):
    if not isinstance(host, str):
        raise ValueError("The Ollama endpoint must be an HTTP URL.")
    parsed = urlparse(host)
    if parsed.scheme not in ("http", "https") or not parsed.hostname or parsed.username or parsed.password:
        raise ValueError("The Ollama endpoint must be an HTTP URL without embedded credentials.")
    if parsed.query or parsed.fragment:
        raise ValueError("The Ollama endpoint cannot contain a query or fragment.")
    return host.rstrip("/")


def _timeout(value):
    if isinstance(value, bool):
        raise ValueError("Backend timeout must be between 1 and 120 seconds.")
    try:
        value = float(value)
    except (TypeError, ValueError, OverflowError) as exc:
        raise ValueError("Backend timeout must be between 1 and 120 seconds.") from exc
    if not math.isfinite(value) or not 1 <= value <= DEFAULT_TIMEOUT:
        raise ValueError("Backend timeout must be between 1 and 120 seconds.")
    return value


def _live_settings(protocol):
    """Use saved numeric settings, but only the collector's implemented prompts."""
    saved = protocol.get("generation_settings")
    settings = dict(DEFAULT_SETTINGS)
    mismatches = []
    if isinstance(saved, dict):
        for key in ("budget", "seed", "temperature", "thinking", "context_window"):
            if key in saved:
                settings[key] = saved[key]
        if saved.get("protocol_version") in ("0.1", "0.2"):
            settings["protocol_version"] = saved["protocol_version"]
        # This validates the actual settings before any inference call.
        _, settings = collection_protocol(settings)
        for key in (*COLLECTION_SETTINGS, "context_window"):
            if key not in saved:
                mismatches.append(f"Saved generation setting {key} is missing.")
            elif saved[key] != settings[key]:
                mismatches.append(f"Saved {key} differs from the live collector protocol.")
        for key in ("initial_seed", "recheck_seed"):
            actual = settings["seed"] if key == "initial_seed" else settings["seed"] + 1
            if key in saved and saved[key] != actual:
                mismatches.append(f"Saved {key} differs from the live collector protocol.")
        # Collector protocols v0.1/v0.2 explicitly execute no finalizer call.
        # The absence of this extra field has that documented version meaning.
        for key in ("finalizer", "finalizer_used"):
            if key in saved and saved[key] is not False:
                mismatches.append("The saved protocol uses a finalizer; live collection does not.")
    else:
        mismatches.append("The released gate does not contain complete generation prompts and settings.")
    protocol_id, settings = collection_protocol(settings)
    if protocol.get("recheck_protocol") != protocol_id:
        mismatches.append("The live collector protocol fingerprint differs from the saved recheck protocol.")
    if protocol.get("action") != "recheck":
        mismatches.append("The saved action is not the collector's explicit recheck action.")
    if not protocol.get("model_digest"):
        mismatches.append("The saved gate does not identify a checkpoint digest.")
    published = protocol.get("published_generation_settings")
    if isinstance(published, dict):
        mismatches.append(f"Released context-window settings are unavailable; live explicitly uses {settings['context_window']} tokens.")
        if published.get("base_generation_limit") != settings["budget"]:
            mismatches.append(f"Released initial generation limit {published.get('base_generation_limit')} differs from live limit {settings['budget']}.")
        if published.get("finalizer_max_tokens"):
            mismatches.append(f"Released calls may use a {published['finalizer_max_tokens']}-token finalizer; live calls execute no finalizer.")
    actual = {**settings, "initial_seed": settings["seed"], "recheck_seed": settings["seed"] + 1,
              "finalizer_used": False, "recheck_protocol": protocol_id}
    return actual, mismatches


def probe_backend(bundle, host=DEFAULT_HOST, timeout=5):
    """Return actual server/model availability and fail closed on solver changes."""
    status = dict(ready=False, status="unavailable", message="", model=None,
                  model_digest=None, experimental=False, protocol_match=False,
                  generation_settings=None, mismatches=[], protocol_fingerprint=None,
                  installed_models=[], live_protocol=None, error_stage=None)
    try:
        # The loader and probe must use the same active class even when only
        # the predictor module was evicted and this live module was retained.
        importlib.import_module("rechecklab.predictor").validate_artifact(bundle)
    except (TypeError, ValueError, KeyError) as exc:
        status.update(status="predictor_load_error", error_stage="predictor",
                      message=f"Saved predictor validation failed: {exc}")
        return status
    if "live_study_protocol" in bundle["protocol"]:
        return _probe_study_backend(bundle, host, timeout, status)
    try:
        host = _host(host)
        timeout = _timeout(timeout)
        protocol = bundle["protocol"]
        status["model"] = protocol["model"]
        settings, mismatches = _live_settings(protocol)
        status["generation_settings"] = settings
        status["mismatches"] = mismatches
    except (TypeError, ValueError) as exc:
        status.update(status="invalid_configuration", error_stage="configuration", message=str(exc))
        return status
    try:
        with urlopen(host + "/api/tags", timeout=timeout) as response:
            payload = json.load(response)
        if not isinstance(payload, dict) or not isinstance(payload.get("models"), list):
            raise ValueError("Ollama did not return a valid installed-model list.")
        installed = {}
        for entry in payload["models"]:
            if not isinstance(entry, dict) or not isinstance(entry.get("name"), str) or not entry["name"].strip():
                raise ValueError("Ollama returned an invalid model identity.")
            digest = entry.get("digest")
            if digest is not None and (not isinstance(digest, str) or not digest.strip()):
                raise ValueError("Ollama returned an invalid checkpoint digest.")
            if entry["name"] in installed and installed[entry["name"]] != digest:
                raise ValueError("Ollama returned conflicting checkpoint identities.")
            installed[entry["name"]] = digest
        status["installed_models"] = sorted(installed)
    except (URLError, TimeoutError, ConnectionError, OSError) as exc:
        status.update(status="unavailable", error_stage="backend", message=f"Ollama is unavailable at {host}: {exc}")
        return status
    except (ValueError, TypeError) as exc:
        status.update(status="invalid_backend_response", error_stage="backend", message=str(exc))
        return status
    model = protocol["model"]
    if model not in installed:
        status.update(status="model_missing", message=f"The saved gate requires {model}. Install that exact model with: ollama pull {model}")
        return status
    digest = installed[model]
    status["model_digest"] = digest
    if not digest:
        status.update(status="checkpoint_unknown", message="Ollama did not identify the installed model's checkpoint digest. Live prediction is blocked.")
        return status
    expected_digest = protocol.get("model_digest")
    if expected_digest is not None and digest != expected_digest:
        status.update(status="checkpoint_mismatch", message="The installed checkpoint digest differs from the saved gate. Live prediction is blocked.")
        return status
    status["protocol_fingerprint"] = protocol_fingerprint(host, model, digest, settings)
    status["live_protocol"] = dict(model=model, model_digest=digest,
                                   recheck_protocol=settings["recheck_protocol"], action="recheck",
                                   generation_settings=collection_protocol(settings)[1],
                                   initial_seed=settings["initial_seed"], recheck_seed=settings["recheck_seed"],
                                   finalizer_used=False)
    is_public = protocol["recheck_protocol"].startswith("sevra:") and "published_generation_settings" in protocol
    if mismatches:
        if not is_public:
            status.update(status="protocol_mismatch", message="The saved gate and live collector settings are incompatible. Use a matching gate before running live.")
            return status
        status.update(ready=True, status="experimental", experimental=True,
                      message=(f"Ollama is ready with {model}. Experimental live use: the public checkpoint digest and full prompts are unavailable. "
                               f"This collector uses a {settings['budget']}-token output limit, {settings['context_window']}-token context, thinking={settings['thinking']}, and no finalizer; its live behavior has not been evaluated by the replay pilot."))
        return status
    status.update(ready=True, status="compatible", protocol_match=True,
                  message=f"Ollama is ready with the saved {model} checkpoint and matching collector settings.")
    return status


def _probe_study_backend(bundle, host, timeout, status):
    """Validate a future study artifact against the exact collection protocol."""
    from .study_protocol import protocol_hash, validate_protocol, verify_backend
    contract = bundle["protocol"]
    try:
        host = _host(host)
        _timeout(timeout)
        protocol = validate_protocol(contract["live_study_protocol"])
        digest = protocol_hash(protocol)
        expected = {"model": protocol["model"], "model_digest": protocol["model_digest"],
                    "action": "recheck", "recheck_protocol": "live-study:" + digest,
                    "study_protocol_hash": digest}
        if any(contract.get(key) != value for key, value in expected.items()):
            raise ValueError("Study artifact identity/hash disagrees with its full frozen protocol.")
        status.update(model=protocol["model"], model_digest=protocol["model_digest"],
                      generation_settings=protocol["generation"], live_protocol=contract,
                      protocol_fingerprint=protocol_fingerprint(host, protocol["model"], protocol["model_digest"], protocol))
    except (TypeError, ValueError, KeyError) as exc:
        status.update(status="protocol_mismatch", error_stage="configuration", message=str(exc))
        return status

        verify_backend(protocol, host)
    except (URLError, TimeoutError, ConnectionError, OSError) as exc:

        return status
    except (ValueError, TypeError, KeyError) as exc:
        status.update(status="protocol_mismatch", error_stage="backend", message=str(exc))
        return status
    status.update(ready=True, protocol_match=True, experimental=True, status="experimental",
                  message="Ollama matches the frozen local-study checkpoint and protocol. Study predictions remain experimental; matching generation settings alone does not establish accuracy.")
    return status


def _answer(content, answer_type):
    if answer_type == "open":
        # Open questions have no trusted key and no task-specific scoring rule.
        # Use only an explicit final marker; never invent an answer from reasoning.
        marked = re.findall(r"\bFINAL(?:\s+ANSWER)?\s*:\s*([^\n]+)", content, re.I)
        return marked[-1].strip().strip(" *`") if marked else ""
    if answer_type not in ("number", "boolean", "choice"):
        raise ValueError("answer_type must be open, number, boolean, or choice.")
    return parse_answer(content, answer_type) or ""


def _call_output(response, model, question, answer_type):
    if not isinstance(response, dict) or response.get("done") is not True:
        raise ValueError("Ollama returned an incomplete response.")
    if response.get("model") != model:
        raise ValueError("Ollama's actual response model differs from the saved gate.")
    message = response.get("message")
    if not isinstance(message, dict) or not isinstance(message.get("content"), str):
        raise ValueError("Ollama did not provide response text.")
    if "thinking" in message and not isinstance(message["thinking"], str):
        raise ValueError("Ollama returned invalid thinking text.")
    if not isinstance(response.get("done_reason"), str) or response["done_reason"].strip().casefold() not in ("stop", "length"):
        raise ValueError("Ollama did not provide a supported actual completion status (stop or length).")
    record = validate_initial_record(dict(
        query=question, base_response=response_text(response),
        base_answer=_answer(message["content"], answer_type),
        base_actual_tokens=actual_tokens(response), base_done_reason=response["done_reason"],
        base_finalizer_used=False,  # This protocol executed no finalizer call.
    ))
    output = dict(response=record["base_response"], answer=record["base_answer"],
                  tokens=int(record["base_actual_tokens"]), done_reason=record["base_done_reason"],
                  finalizer_used=False, parse_valid=bool(record["base_answer"]),
                  prompt_eval_count=int(float(response["prompt_eval_count"])),
                  eval_count=int(float(response["eval_count"])),
                  model=model, wall_seconds=response.get("wall_seconds"))
    return record, output, message["content"] or record["base_response"]


def _reported_attempt(response):
    """Retain returned text/metadata for failure audit without validating it as input."""
    if not isinstance(response, dict):
        return {"response": None, "reported_metadata": {"response_type": type(response).__name__},
                "validated": False}
    message = response.get("message")
    text = None
    if isinstance(message, dict) and isinstance(message.get("content"), str):
        text = message["content"]
        if isinstance(message.get("thinking"), str) and message["thinking"]:
            text = message["thinking"] + "\n\n" + text
    # Repr preserves missing/invalid token metadata without emitting NaN JSON.
    metadata = {key: repr(response[key]) if key in response else "<missing>"
                for key in ("model", "done", "done_reason", "prompt_eval_count", "eval_count")}
    return {"response": text, "reported_metadata": metadata, "validated": False}


def run_live_question(question, bundle, host=DEFAULT_HOST, timeout=DEFAULT_TIMEOUT,
                      allow_experimental=False, on_progress=None, answer_type="open"):
    """Generate an initial answer, predict using initial signals, then obey the gate.

No trusted key can be passed into this function. Correctness and the actual
HELP/HURT/NO_CHANGE outcome remain unknown without separate offline scoring.
"""
    result = dict(status="blocked", initial=None, probabilities=None, decision=None,
                  recheck_attempted=False, rechecked=False, revised=None,
                  selected_answer=None, selected_response=None, selected_source=None,
                  error=None, error_stage=None, backend=None, live_protocol=None, initial_record=None,
                  initial_attempt=None, recheck_attempt=None,
                  actual_outcome="Unknown", correctness="Unknown")

    def progress(stage, message):
        if on_progress is not None:
            on_progress(stage, message)

    if not isinstance(question, str) or not question.strip():
        result.update(status="input_error", error="Enter a nonempty question.")
        return result
    try:
        timeout = _timeout(timeout)
        if answer_type not in ("open", "number", "boolean", "choice"):
            raise ValueError("answer_type must be open, number, boolean, or choice.")
    except ValueError as exc:
        result.update(status="input_error", error=str(exc))
        return result
    progress("checking_backend", "Checking the actual solver and saved gate compatibility.")
    backend = probe_backend(bundle, host=host)
    result["backend"] = backend
    result["live_protocol"] = backend.get("live_protocol")
    if not backend["ready"]:
        result["error"] = backend["message"]
        result["error_stage"] = backend.get("error_stage")
        if backend["status"] == "predictor_load_error":
            result["status"] = "predictor_load_error"
        progress("failed", result["error"])
        return result
    if backend["experimental"] and not allow_experimental:
        result["error"] = "This public gate requires explicitly identified experimental live use before inference."
        progress("failed", result["error"])
        return result
    if "live_study_protocol" in bundle["protocol"]:
        return _run_study_question(question, answer_type, bundle, backend, host, timeout, progress, result)
    settings, model = backend["generation_settings"], backend["model"]
    messages = [dict(role="user", content=INITIAL_INSTRUCTION + "\n\n" + question)]
    progress("generating_initial", "Generating the initial answer with the saved solver.")
    try:
        initial_response = ollama_chat(model, messages, host, settings["budget"], settings["initial_seed"],
                                       settings["temperature"], settings["thinking"], timeout=timeout,
                                       context_window=settings["context_window"])
        result["initial_attempt"] = _reported_attempt(initial_response)
        record, initial, previous = _call_output(initial_response, model, question, answer_type)
        result["initial_attempt"]["validated"] = True
        actual_identity = {key: backend["live_protocol"][key] for key in ("model", "recheck_protocol", "action")}
        if backend["model_digest"] is not None:
            actual_identity["model_digest"] = backend["model_digest"]
        result["initial_record"] = {**record, **actual_identity}
        result["initial"] = initial
        result.update(selected_answer=initial["answer"], selected_response=initial["response"], selected_source="initial")
    except (URLError, TimeoutError, ConnectionError, OSError, TypeError, ValueError, KeyError) as exc:
        if result["initial_attempt"] is None and hasattr(exc, "response"):
            result["initial_attempt"] = _reported_attempt(exc.response)
        result.update(status="initial_failed", error_stage="backend", error=f"Initial generation failed: {exc}")
        progress("failed", result["error"])
        return result
    progress("predicting", "Estimating HELP, NO_CHANGE, and HURT from the initial response only.")
    try:
        # Backend identity was checked above. A public experimental run keeps
        # its actual protocol separate instead of relabeling it as the replay.
        if backend["protocol_match"]:
            frame = pd.DataFrame([result["initial_record"]])
        else:
            frame = pd.DataFrame([{**{key: record[key] for key in INITIAL_REQUIRED}, "model": model}])
        probabilities = bundle["predictor"].probabilities(frame)
        recheck = bool(decide_recheck(probabilities, bundle["threshold"], bundle["harm_weight"])[0])
        result["probabilities"] = probabilities[0].tolist()
        result["decision"] = "Recheck" if recheck else "Skip recheck"
    except (TypeError, ValueError, KeyError) as exc:
        result.update(status="prediction_failed", error_stage="predictor", error=f"Prediction failed; the initial answer is retained: {exc}")
        progress("failed", result["error"])
        return result
    if not recheck:
        result.update(status="complete", recheck_tokens=0, total_tokens=initial["tokens"])
        progress("complete", "The gate skipped the recheck and retained the initial answer. Correctness is unknown.")
        return result
    result["recheck_attempted"] = True
    progress("rechecking", "The gate chose one recheck; checking the fixed initial answer.")
    try:
        revision_response = ollama_chat(model, messages + [dict(role="assistant", content=previous),
                                       dict(role="user", content=RECHECK_INSTRUCTION)], host,
                                       settings["budget"], settings["recheck_seed"], settings["temperature"],
                                       settings["thinking"], timeout=timeout, context_window=settings["context_window"])
        result["recheck_attempt"] = _reported_attempt(revision_response)
        _, revised, _ = _call_output(revision_response, model, question, answer_type)
        result["recheck_attempt"]["validated"] = True
        result.update(status="complete", revised=revised, rechecked=True,
                      selected_answer=revised["answer"], selected_response=revised["response"], selected_source="revised",
                      recheck_tokens=revised["tokens"], total_tokens=initial["tokens"] + revised["tokens"])
        progress("complete", "One recheck completed. Its correctness and outcome require a trusted answer key.")
    except (URLError, TimeoutError, ConnectionError, OSError, TypeError, ValueError, KeyError) as exc:
        if result["recheck_attempt"] is None and hasattr(exc, "response"):
            result["recheck_attempt"] = _reported_attempt(exc.response)
        result.update(status="recheck_failed", selected_source="initial_fallback",
                      error_stage="backend",
                      recheck_tokens=None, total_tokens=None,
                      error=f"The gate chose a recheck, but it failed; the initial answer is retained: {exc}")
        progress("failed", result["error"])
    return result


def _run_study_question(question, answer_type, bundle, backend, host, timeout, progress, result):
    """Use exactly the same call, parser and feature record as offline collection."""
    from .study_protocol import execute_call
    protocol = bundle["protocol"]["live_study_protocol"]
    if answer_type == "boolean":
        result.update(status="input_error", error="This local-study protocol supports numeric, choice or open questions.")
        return result
    started = time.perf_counter()
    progress("generating_initial", "Generating under the frozen local-study protocol.")
    try:
        raw, record, initial = execute_call(protocol, question, answer_type, "initial", host=host, timeout=timeout)
        result["initial_attempt"] = {**_reported_attempt(raw), "validated": True}
        identity = {key: bundle["protocol"][key] for key in ("model", "model_digest", "action", "recheck_protocol", "study_protocol_hash")}
        result.update(initial=initial, initial_record={**record, **identity}, selected_answer=initial["answer"],
                      selected_response=initial["response"], selected_source="initial")
    except (URLError, TimeoutError, ConnectionError, OSError, TypeError, ValueError, KeyError) as exc:
        result.update(status="initial_failed", error_stage="backend", error=f"Study initial generation failed: {exc}")
        if hasattr(exc, "response"):
            result["initial_attempt"] = _reported_attempt(exc.response)
        progress("failed", result["error"])
        return result
    progress("predicting", "Estimating recheck effects from initial signals only.")
    prediction_started = time.perf_counter()
    try:
        probabilities = bundle["predictor"].probabilities(pd.DataFrame([result["initial_record"]]))
        recheck = bool(decide_recheck(probabilities, bundle["threshold"], bundle["harm_weight"])[0])
        result.update(probabilities=probabilities[0].tolist(), decision="Recheck" if recheck else "Skip recheck",
                      predictor_wall_seconds=time.perf_counter() - prediction_started)
    except (TypeError, ValueError, KeyError) as exc:
        result.update(status="prediction_failed", error_stage="predictor", error=f"Study prediction failed; initial answer retained: {exc}")
        progress("failed", result["error"])
        return result
    if not recheck:
        result.update(status="complete", recheck_tokens=0, total_tokens=initial["tokens"],
                      live_generation_prediction_wall_seconds=time.perf_counter() - started)
        progress("complete", "Skipped recheck; correctness remains unknown.")
        return result
    result["recheck_attempted"] = True
    progress("rechecking", "Rechecking the same question and full saved initial response.")
    try:
        raw, _, revised = execute_call(protocol, question, answer_type, "recheck", initial["response"], host=host, timeout=timeout)
        result["recheck_attempt"] = {**_reported_attempt(raw), "validated": True}
        result.update(status="complete", revised=revised, rechecked=True, selected_answer=revised["answer"],
                      selected_response=revised["response"], selected_source="revised", recheck_tokens=revised["tokens"],
                      total_tokens=initial["tokens"] + revised["tokens"],
                      live_generation_prediction_wall_seconds=time.perf_counter() - started)
        progress("complete", "One protocol-matched recheck completed; actual correctness requires a separate trusted key.")
    except (URLError, TimeoutError, ConnectionError, OSError, TypeError, ValueError, KeyError) as exc:
        result.update(status="recheck_failed", selected_source="initial_fallback", error_stage="backend",
                      error=f"Study recheck failed; initial answer retained: {exc}",
                      live_generation_prediction_wall_seconds=time.perf_counter() - started)
        if hasattr(exc, "response"):
            result["recheck_attempt"] = _reported_attempt(exc.response)
        progress("failed", result["error"])
    return result

"""Session-scoped live call deduplication, including overlapping Streamlit reruns."""
from __future__ import annotations

import hashlib
import json
from threading import Lock


def live_input_key(question, model_signature, host, backend):
    payload = dict(question=question.strip(), artifact=model_signature,
                   host=host.rstrip("/"), protocol=backend.get("protocol_fingerprint"),
                   model=backend.get("model"), digest=backend.get("model_digest"),
                   settings=backend.get("generation_settings"))
    return hashlib.sha256(json.dumps(payload, sort_keys=True, allow_nan=False).encode("utf-8")).hexdigest()


def execute_once(state, key, operation):
    """Cache completed or failed jobs; a busy session never starts another call.

    Input-change callbacks clear the displayed result, not these jobs or lock.
    A failed attempt is retained too: retrying a button must not repeat an
    initial call whose recheck may already have started on the server.
    """
    lock = state.setdefault("_live_lock", Lock())
    if not lock.acquire(blocking=False):
        return {"status": "busy"}
    try:
        jobs = state.setdefault("_live_jobs", {})
        if key in jobs:
            return jobs[key]
        # A Streamlit stop/rerun can interrupt a script after the server has
        # received its request. Keep a tombstone rather than risk replaying it.
        jobs[key] = {"status": "busy"}
        try:
            job = {"status": "complete", "result": operation()}
        except Exception as exc:
            job = {"status": "failed", "error": f"Live execution failed: {exc}"}
        jobs[key] = job
        # Retain every request in this browser session, including failed calls.
        # Eviction could make returning to a previous question call the LLM twice.
        return job
    finally:
        lock.release()

"""Question-equal prediction/policy metrics and question bootstrap intervals."""
import numpy as np
import pandas as pd
from sklearn.metrics import confusion_matrix, precision_recall_fscore_support

from .data import LABELS


def question_weights(frame):
    if frame.empty or "group_id" not in frame or frame.group_id.isna().any():
        raise ValueError("Metrics require nonempty, validated question groups.")
    return 1.0 / frame.groupby("group_id").group_id.transform("size").to_numpy()


def validate_decisions(frame, decisions):
    decisions = np.asarray(decisions)
    if decisions.shape != (len(frame),) or decisions.dtype != np.dtype(bool):
        raise ValueError("Decisions must be one explicit boolean per outcome row.")
    return decisions


def policy_metrics(frame, decisions):
    decisions = validate_decisions(frame, decisions)
    question_weights(frame)
    base = frame.base_correct.to_numpy(dtype=float)
    final = np.where(decisions, frame.final_correct.to_numpy(dtype=float), base)
    per_row = pd.DataFrame({
        "group_id": frame.group_id.to_numpy(),
        "accuracy": final, "base_correct": base,
        "help": decisions * (frame.label == "HELP").to_numpy(),
        "hurt": decisions * (frame.label == "HURT").to_numpy(),
        "recheck": decisions.astype(float),
        "base_tokens": frame.base_actual_tokens.to_numpy(dtype=float),
        "extra_tokens": decisions * frame.action_actual_tokens.to_numpy(dtype=float),
        "all_extra_tokens": frame.action_actual_tokens.to_numpy(dtype=float),
    })
    grouped = per_row.groupby("group_id").mean(numeric_only=True)
    result = {"questions": len(grouped), "rollout_rows": len(frame),
        "accuracy_pct": 100 * grouped.accuracy.mean(),
        "harmful_flip_pct_of_all_questions": 100 * grouped.hurt.mean(),
        "helpful_fix_pct_of_all_questions": 100 * grouped.help.mean(),
        "harm_pct_of_initially_correct": 100 * grouped.hurt.sum() / grouped.base_correct.sum() if grouped.base_correct.sum() else None,
        "recovery_pct_of_initially_wrong": 100 * grouped.help.sum() / (1-grouped.base_correct).sum() if (1-grouped.base_correct).sum() else None,
        "recheck_pct": 100 * grouped.recheck.mean(),
        "average_total_tokens": (grouped.base_tokens + grouped.extra_tokens).mean(),
        "average_extra_tokens": grouped.extra_tokens.mean(),
        "saved_extra_tokens_pct": 100 * (1 - grouped.extra_tokens.sum()/grouped.all_extra_tokens.sum()) if grouped.all_extra_tokens.sum() else 0,
        "saved_total_tokens_pct": 100 * (1 - (grouped.base_tokens+grouped.extra_tokens).sum()/(grouped.base_tokens+grouped.all_extra_tokens).sum())
            if (grouped.base_tokens+grouped.all_extra_tokens).sum() else 0,
    }
    return {k: (float(v) if isinstance(v, (np.floating, np.integer)) else v) for k, v in result.items()}


def _macro_from_matrix(matrix):
    diagonal = np.diag(matrix)
    denominator = matrix.sum(axis=0) + matrix.sum(axis=1)
    return float(np.divide(2*diagonal, denominator, out=np.zeros(3), where=denominator != 0).mean())


def predictor_metrics(frame, probabilities, bootstrap_repeats=0, seed=42):
    from .predictor import validate_probabilities
    probabilities = validate_probabilities(probabilities, len(frame))
    weights = question_weights(frame)
    truth = frame.label.to_numpy()
    if not set(truth).issubset(LABELS):
        raise ValueError("Unknown outcome labels in evaluation data.")
    predicted = np.array(LABELS)[probabilities.argmax(axis=1)]
    precision, recall, f1, support = precision_recall_fscore_support(
        truth, predicted, labels=LABELS, sample_weight=weights, zero_division=0)
    indices = np.array([LABELS.index(y) for y in truth])
    losses = -np.log(np.clip(probabilities[np.arange(len(frame)), indices], 1e-15, 1))
    loss = float(np.average(losses, weights=weights))
    result = {
        "macro_f1": float(f1.mean()), "log_loss": loss, "multiclass_log_loss": loss,
        "accuracy": float(np.average(truth == predicted, weights=weights)),
        "per_class": {label: {"precision": float(precision[i]), "recall": float(recall[i]),
            "f1": float(f1[i]), "question_weighted_support": float(support[i])} for i, label in enumerate(LABELS)},
        "confusion_matrix": confusion_matrix(truth, predicted, labels=LABELS, sample_weight=weights).tolist(),
        "row_confusion_matrix": confusion_matrix(truth, predicted, labels=LABELS).tolist(),
        "label_order": list(LABELS), "class_counts": {k: int(v) for k,v in frame.label.value_counts().items()},
        "predicted_class_counts": {k: int((predicted == k).sum()) for k in LABELS},
        "questions": int(frame.group_id.nunique()), "rollout_rows": len(frame),
        "aggregation": "Each question has weight 1; its outcome rows share that weight.",
    }
    if bootstrap_repeats:
        codes, groups = pd.factorize(frame.group_id, sort=True)
        matrices = np.zeros((len(groups), 3, 3))
        pred_indices = probabilities.argmax(axis=1)
        for row, group in enumerate(codes):
            matrices[group, indices[row], pred_indices[row]] += weights[row]
        group_losses = pd.DataFrame({"group": codes, "loss": losses}).groupby("group").loss.mean().to_numpy()
        rng = np.random.default_rng(seed)
        draws = rng.integers(0, len(groups), size=(bootstrap_repeats, len(groups)))
        fs = [_macro_from_matrix(matrices[draw].sum(axis=0)) for draw in draws]
        result["macro_f1_95pct_ci"] = np.quantile(fs, [.025, .975]).tolist()
        result["log_loss_95pct_ci"] = np.quantile(group_losses[draws].mean(axis=1), [.025, .975]).tolist()
        result["uncertainty"] = "Question bootstrap conditional on this fixed fitted model and test partition."
    return result


def paired_accuracy_interval(frame, decisions, repeats=600, seed=42):
    decisions = validate_decisions(frame, decisions)
    diff = np.where(decisions, frame.final_correct, frame.base_correct).astype(float) - frame.base_correct.to_numpy(dtype=float)
    means = pd.DataFrame({"group": frame.group_id.to_numpy(), "difference": diff}).groupby("group").difference.mean().to_numpy()
    rng = np.random.default_rng(seed)
    sampled = rng.choice(means, size=(repeats, len(means)), replace=True).mean(axis=1)
    return [float(x) for x in np.quantile(100*sampled, [0.025, 0.975])]

def validate_probabilities(probabilities, expected_rows=None):
    """Check explicit HELP, NO_CHANGE, HURT columns; never repair invalid output."""
    try:
        result = np.asarray(probabilities, dtype=float)
    except (TypeError, ValueError) as exc:
        raise ValueError("Probabilities must be a numeric matrix in the declared class order.") from exc
    if result.ndim != 2 or result.shape[1] != len(LABELS):
        raise ValueError(f"Probabilities must have shape (rows, {len(LABELS)}) in class order {LABELS}.")
    if expected_rows is not None and result.shape[0] != expected_rows:
        raise ValueError("Probability row count does not match the initial inputs.")
    if not np.isfinite(result).all() or (result < 0).any() or (result > 1).any():
        raise ValueError("Probabilities must be finite numbers between zero and one.")
    if not np.allclose(result.sum(axis=1), 1.0, rtol=1e-7, atol=1e-8):
        raise ValueError("Outcome probabilities must sum to one in every row.")
    return result


def _positive_finite(value, name, allow_zero=False):
    try:
        number = float(value)
    except (ValueError, TypeError, OverflowError) as exc:
        raise ValueError(f"{name} must be finite and {'nonnegative' if allow_zero else 'positive'}.") from exc
    if not math.isfinite(number) or (number < 0 if allow_zero else number <= 0):
        raise ValueError(f"{name} must be finite and {'nonnegative' if allow_zero else 'positive'}.")
    return number


def _finite_threshold(value):
    try:
        threshold = float(value)
    except (ValueError, TypeError, OverflowError) as exc:
        raise ValueError("The recheck threshold must be finite.") from exc
    if not math.isfinite(threshold):
        raise ValueError("The recheck threshold must be finite.")
    return threshold


def _protocol_contract(protocol):
    if not isinstance(protocol, dict):
        raise ValueError("A model and recheck-protocol compatibility dictionary is required.")
    for key in ("model", "recheck_protocol", "action"):
        if not isinstance(protocol.get(key), str) or not protocol[key].strip():
            raise ValueError(f"The compatibility contract requires a nonempty {key}.")
    if "model_digest" in protocol and (
        not isinstance(protocol["model_digest"], str) or not protocol["model_digest"].strip()
    ):
        raise ValueError("model_digest must be nonempty when supplied.")
    return dict(protocol)


@dataclass
class OutcomePredictor:
    estimator: object
    model_name: str
    feature_schema_version: str = FEATURE_SCHEMA_VERSION
    feature_schema_fingerprint: str = FEATURE_SCHEMA_FINGERPRINT
    feature_names: tuple = field(default_factory=lambda: tuple(FEATURE_NAMES))
    class_order: tuple = field(default_factory=lambda: tuple(LABELS))
    protocol: dict | None = None

    def validate_contract(self):
        if (getattr(self, "feature_schema_version", None) != FEATURE_SCHEMA_VERSION or
            getattr(self, "feature_schema_fingerprint", None) != FEATURE_SCHEMA_FINGERPRINT or
            list(getattr(self, "feature_names", ())) != FEATURE_NAMES or
            list(getattr(self, "class_order", ())) != LABELS):
            raise ValueError("The predictor uses an incompatible feature schema or class order; rerun the pilot.")
        if isinstance(self.estimator, BaseEstimator):
            try:
                check_is_fitted(self.estimator)
            except (NotFittedError, TypeError) as exc:
                raise ValueError("The predictor's scikit-learn estimator is unfitted.") from exc
        classes = np.asarray(getattr(self.estimator, "classes_", []))
        if (classes.ndim != 1 or not len(classes) or len(set(classes.tolist())) != len(classes) or
            any(name not in LABELS for name in classes)):
            raise ValueError("The estimator must declare unique, recognized outcome classes.")
        count = getattr(self.estimator, "n_features_in_", None)
        if count is not None and count != len(FEATURE_NAMES):
            raise ValueError("The estimator was fitted with an incompatible number of features.")
        fitted_names = getattr(self.estimator, "feature_names_in_", None)
        if fitted_names is not None and list(fitted_names) != FEATURE_NAMES:
            raise ValueError("The estimator's fitted feature ordering is incompatible.")
        if self.protocol is not None:
            _protocol_contract(self.protocol)

    def probabilities(self, frame):
        self.validate_contract()
        # Metadata present in a recorded dataset must agree with the saved solver/action.
        if self.protocol is not None:
            for key, expected in self.protocol.items():
                if key in frame.columns and not frame[key].eq(expected).all():
                    raise ValueError(f"Initial records have incompatible {key} metadata.")
        features = extract_features(frame)
        try:
            raw = np.asarray(self.estimator.predict_proba(features), dtype=float)
        except (TypeError, ValueError) as exc:
            raise ValueError(f"The estimator could not produce valid outcome probabilities: {exc}") from exc
        classes = list(self.estimator.classes_)
        if raw.ndim != 2 or raw.shape != (len(frame), len(classes)):
            raise ValueError("Estimator probability columns do not match its declared classes.")
        result = np.zeros((len(frame), len(LABELS)), dtype=float)
        for index, name in enumerate(classes):
            result[:, LABELS.index(name)] = raw[:, index]
        return validate_probabilities(result, len(frame))


def make_artifact(predictor, threshold, harm_weight=1.0, protocol=None):
    protocol = _protocol_contract(protocol)
    predictor.protocol = protocol
    bundle = {
        "artifact_schema_version": ARTIFACT_SCHEMA_VERSION,
        "feature_schema_version": FEATURE_SCHEMA_VERSION,
        "feature_schema_fingerprint": FEATURE_SCHEMA_FINGERPRINT,
        "feature_names": list(FEATURE_NAMES), "class_order": list(LABELS),
        "protocol": protocol, "predictor": predictor,
        "threshold": _finite_threshold(threshold),
        "harm_weight": _positive_finite(harm_weight, "harm_weight"),
    }
    return validate_artifact(bundle)


def validate_artifact(bundle, expected_protocol=None):
    if not isinstance(bundle, dict):
        raise ValueError("The saved gate must contain a versioned compatibility contract.")
    if (bundle.get("artifact_schema_version") != ARTIFACT_SCHEMA_VERSION or
        bundle.get("feature_schema_version") != FEATURE_SCHEMA_VERSION or
        bundle.get("feature_schema_fingerprint") != FEATURE_SCHEMA_FINGERPRINT or
        bundle.get("feature_names") != FEATURE_NAMES or bundle.get("class_order") != LABELS):
        raise ValueError("The saved gate has an incompatible artifact, feature schema, or class order; rerun the pilot.")
    predictor = bundle.get("predictor")
    if not isinstance(predictor, OutcomePredictor):
        raise ValueError("The saved gate must contain an OutcomePredictor.")
    predictor.validate_contract()
    protocol = _protocol_contract(bundle.get("protocol"))
    if predictor.protocol != protocol:
        raise ValueError("The predictor and artifact disagree about the model/recheck protocol.")
    if expected_protocol is not None and protocol != _protocol_contract(expected_protocol):
        raise ValueError("The saved gate is incompatible with the displayed experiment's model/recheck protocol.")
    _finite_threshold(bundle.get("threshold"))
    _positive_finite(bundle.get("harm_weight"), "harm_weight")
    return bundle


def save_artifact(bundle, path):
    """Write the production plain-dictionary format after strict validation."""
    # Resolve the active module after a file-watcher reload; this retains the
    # exact class check without accepting a similarly named foreign object.
    importlib.import_module(__name__).validate_artifact(bundle)
    path = Path(path).resolve()
    path.parent.mkdir(parents=True, exist_ok=True)
    joblib.dump(bundle, path)
    return path


def _artifact_description(value):
    value_type = type(value)
    description = f"loaded_type={value_type.__module__}.{value_type.__qualname__}"
    if isinstance(value, dict):
        description += "; keys=" + repr(sorted(str(key) for key in value))
        predictor_type = type(value.get("predictor"))
        description += f"; predictor_type={predictor_type.__module__}.{predictor_type.__qualname__}"
    return description


def load_artifact(path, *, expected_protocol=None, expected_sha256=None):
    """Read exact artifact bytes and validate against the active canonical class.

    Streamlit can retain an imported helper after reloading local modules.
    Joblib unpickles through the current module, so use that module's validator
    as well. Model/schema/protocol checks remain strict; no retraining or type
    coercion is performed.
    """
    path = Path(path).resolve()
    loaded = None
    description = "loaded_type=<not loaded>"
    expected_import = "rechecklab.predictor.OutcomePredictor"
    try:
        raw = path.read_bytes()
        if expected_sha256 is not None:
            if (not isinstance(expected_sha256, str) or len(expected_sha256) != 64 or
                any(character not in "0123456789abcdefABCDEF" for character in expected_sha256)):
                raise ValueError("Expected artifact SHA-256 must be a 64-character hexadecimal string.")
            if hashlib.sha256(raw).hexdigest() != expected_sha256.lower():
                raise ValueError("Artifact SHA-256 does not match the displayed experiment.")
        loaded = joblib.load(io.BytesIO(raw))
        description = _artifact_description(loaded)
        active_module = importlib.import_module(__name__)
        active_module.validate_artifact(loaded, expected_protocol=expected_protocol)
    except Exception as exc:
        raise ArtifactLoadError(f"Cannot load saved gate at {path}: {exc} [{description}; expected_import={expected_import}]") from exc
    return loaded


def _question_weights(frame):
    if frame.empty or "group_id" not in frame or frame.group_id.isna().any():
        raise ValueError("Model fitting and threshold selection require nonempty question-grouped records.")
    # Each question has total training weight one, regardless of its rollouts.
    return 1.0 / frame.groupby("group_id")["group_id"].transform("size").to_numpy()


def fit_predictor(train, dev, seed=42, return_candidates=False):
    """Compare a majority baseline and balanced/unbalanced models on source dev."""
    from .metrics import predictor_metrics

    if "label" not in train or "label" not in dev or train.label.nunique() < 2:
        raise ValueError("Training needs at least two outcome classes. Collect more paired responses.")
    if train.empty or dev.empty or not train.label.isin(LABELS).all() or not dev.label.isin(LABELS).all():
        raise ValueError("Source training and development require nonempty valid outcome labels.")
    _question_weights(train)
    _question_weights(dev)
    if set(train.group_id) & set(dev.group_id):
        raise ValueError("Source training and development questions must be disjoint.")
    x_train = extract_features(train)
    extract_features(dev)
    question_weights = _question_weights(train)
    present = list(train.label.unique())
    class_mass = {label: question_weights[train.label.to_numpy() == label].sum() for label in present}
    balanced_weights = question_weights * np.array([
        question_weights.sum() / (len(present) * class_mass[label]) for label in train.label
    ])
    candidates = [
        ("majority_class", DummyClassifier(strategy="prior"), False),
        ("logistic_regression", make_pipeline(StandardScaler(), LogisticRegression(C=0.5, max_iter=1500, random_state=seed)), False),
        ("logistic_regression_balanced", make_pipeline(StandardScaler(), LogisticRegression(C=0.5, max_iter=1500, random_state=seed)), True),
        ("hist_gradient_boosting", HistGradientBoostingClassifier(max_iter=100, max_leaf_nodes=7,
            l2_regularization=5, learning_rate=0.05, random_state=seed), False),
        ("hist_gradient_boosting_balanced", HistGradientBoostingClassifier(max_iter=100, max_leaf_nodes=7,
            l2_regularization=5, learning_rate=0.05, random_state=seed), True),
    ]
    scores, fitted = [], {}
    for name, estimator, balanced in candidates:
        weights = balanced_weights if balanced else question_weights
        if hasattr(estimator, "named_steps"):
            estimator.fit(x_train, train.label, standardscaler__sample_weight=question_weights,
                          logisticregression__sample_weight=weights)
        else:
            estimator.fit(x_train, train.label, sample_weight=weights)
        predictor = OutcomePredictor(estimator, name)
        # Build via the same public path used for evaluation and the demo.
        probabilities = predictor.probabilities(dev)
        metrics = predictor_metrics(dev, probabilities)
        loss = metrics.get("log_loss", metrics.get("multiclass_log_loss"))
        if loss is None:
            indices = np.array([LABELS.index(label) for label in dev.label])
            loss = np.average(-np.log(np.clip(probabilities[np.arange(len(dev)), indices], 1e-15, 1)),
                              weights=_question_weights(dev))
        scores.append({"model": name, "class_balanced_training": balanced,
                       "source_dev_macro_f1": float(metrics["macro_f1"]),
                       "source_dev_log_loss": float(loss), "source_dev_metrics": metrics,
                       "selection_rule": MODEL_SELECTION_RULE})
        fitted[name] = predictor
    winner = max(range(len(scores)), key=lambda index: (
        round(scores[index]["source_dev_macro_f1"], 12),
        -round(scores[index]["source_dev_log_loss"], 12), -index,
    ))
    for index, row in enumerate(scores):
        row["selected"] = index == winner
    result = (fitted[scores[winner]["model"]], scores)
    return (*result, fitted) if return_candidates else result


def revision_score(probabilities, harm_weight=1.0):
    probabilities = validate_probabilities(probabilities)
    harm_weight = _positive_finite(harm_weight, "harm_weight")
    return probabilities[:, LABELS.index("HELP")] - harm_weight * probabilities[:, LABELS.index("HURT")]


def decide_recheck(probabilities, threshold, harm_weight=1.0):
    """A decision to skip keeps the initial answer; it does not certify correctness."""
    return revision_score(probabilities, harm_weight) > _finite_threshold(threshold)


def choose_threshold(frame, scores, harm_weight=1.0, token_penalty=0.0):
    """Choose on development/adaptation labels; gold is never an inference input."""
    harm_weight = _positive_finite(harm_weight, "harm_weight")
    token_penalty = _positive_finite(token_penalty, "token_penalty", allow_zero=True)
    scores = np.asarray(scores, dtype=float)
    if scores.ndim != 1 or len(scores) != len(frame) or not len(scores) or not np.isfinite(scores).all():
        raise ValueError("Threshold selection requires one finite score per development question rollout.")
    if "label" not in frame or not frame.label.isin(LABELS).all():
        raise ValueError("Threshold selection requires valid development outcome labels.")
    weights = _question_weights(frame)
    if "action_actual_tokens" not in frame:
        raise ValueError("Threshold selection requires recorded recheck token counts.")
    try:
        cost = frame.action_actual_tokens.to_numpy(dtype=float)
    except (TypeError, ValueError) as exc:
        raise ValueError("Invalid recorded recheck token counts.") from exc
    if not np.isfinite(cost).all() or (cost < 0).any():
        raise ValueError("Invalid recorded recheck token counts.")
    # Every observed score is a candidate, so repeated rollouts cannot bias a
    # quantile grid or hide a distinct decision boundary. Strict > is shared.
    thresholds = np.unique(np.r_[np.nextafter(scores.min(), -np.inf), scores, np.nextafter(scores.max(), np.inf)])
    thresholds = thresholds[np.isfinite(thresholds)]
    gain = (frame.label == "HELP").to_numpy().astype(float) - harm_weight * (frame.label == "HURT").to_numpy()
    normalized_cost = cost / max(float(np.average(cost, weights=weights)), 1)
    best = None
    for threshold in thresholds:
        decisions = scores > threshold
        utility = float(np.average(decisions * (gain - token_penalty * normalized_cost), weights=weights))
        used_tokens = float(np.average(decisions * cost, weights=weights))
        key = (round(utility, 12), -round(used_tokens, 12), float(threshold))
        if best is None or key > best[0]:
            best = (key, float(threshold))
    return best[1]

def read_jsonl(path):
    path = Path(path)
    if not path.exists():
        return []
    records = []
    # JSONL uses literal LF. str.splitlines() also splits Unicode NEL/LS/PS
    # characters embedded in real benchmark question strings.
    for line_number, line in enumerate(path.read_text(encoding="utf-8").split("\n"), 1):
        if line.strip():
            try:
                records.append(json.loads(line))
            except ValueError as exc:
                raise ValueError(f"Invalid durable journal {path}:{line_number}; repair explicitly before resuming.") from exc
    return records

def audit_safe(value):
    """Retain rejected nonfinite backend metadata as explicit failure evidence."""
    if isinstance(value, dict):
        return {key: audit_safe(item) for key, item in value.items()}
    if isinstance(value, list):
        return [audit_safe(item) for item in value]
    if isinstance(value, float) and not math.isfinite(value):
        return {"invalid_nonfinite_float": repr(value)}
    return value


def validate_manifest(questions, keys):
    if not questions or len({q["example_id"] for q in questions}) != len(questions):
        raise ValueError("Unique preselected question IDs are required.")
    if len({q["group_id"] for q in questions}) != len(questions):
        raise ValueError("Duplicate/overlapping normalized question groups in manifest.")
    keymap = {k["example_id"]: k for k in keys}
    if len(keymap) != len(keys) or set(keymap) != {q["example_id"] for q in questions}:
        raise ValueError("Separate trusted keys must exactly match the frozen questions.")
    for q in questions:
        if any(k in q for k in ("gold_answer", "answerKey", "label", "base_correct", "final_correct")):
            raise ValueError("Question manifest must not contain answer keys or outcomes.")

        if q["partition"] not in ("source_train", "source_development", "source_test", "target_adaptation_pool", "target_test"):
            raise ValueError("Unknown predeclared question partition.")
        grade_answer(None, keymap[q["example_id"]]["gold_answer"], q["answer_type"])

def journal_state(events, questions, digest, protocol=None):
    identities = {q["example_id"]: q for q in questions}
    started, terminals, successes, failures = {}, {}, {}, []
    for event in events:
        if event.get("protocol_hash") != digest:
            raise ValueError("Journal mixes protocols; collection cannot resume.")
        example = event.get("example_id")
        if example not in identities or event.get("group_id") != identities[example]["group_id"]:
            raise ValueError("Journal call does not match frozen manifest.")
        if event.get("phase") not in ("initial", "recheck"):
            raise ValueError("Unknown call phase in journal.")
        cid = event["call_id"]
        if event["status"] == "started":
            if cid in started:
                raise ValueError("Duplicate started call journal event.")
            started[cid] = event
        elif event["status"] in ("succeeded", "failed", "resolved_uncertain", "cancelled_no_request"):
            if cid not in started or cid in terminals:
                raise ValueError("Unmatched or duplicate terminal call journal event.")
            origin = started[cid]
            for key in ("example_id", "group_id", "phase", "seed", "attempt", "session_id", "protocol_hash"):
                if event.get(key) != origin.get(key):
                    raise ValueError(f"Terminal call changed its started {key} identity.")
            terminals[cid] = event
            if event["status"] == "succeeded":
                if protocol is not None:
                    question = identities[example]
                    _, actual = normalize_call(protocol, question["query"], question["answer_type"], event["response"])
                    if event.get("output") != actual or event.get("seed") != protocol["generation"][event["phase"] + "_seed"]:
                        raise ValueError("Saved successful call output/seed disagrees with its actual response or frozen protocol.")
                key = (example, event["phase"])
                if key in successes:
                    raise ValueError("Multiple successful calls for one question/phase.")
                successes[key] = event
            elif event["status"] != "cancelled_no_request":
                failures.append(event)
        else:
            raise ValueError("Unknown journal status.")
    pending = [event for cid, event in started.items() if cid not in terminals]
    return successes, failures, pending


def build_pair(question, key, initial, recheck, protocol, digest):
    base, final = initial["output"], recheck["output"]
    base_correct = grade_answer(base["answer"] or None, key["gold_answer"], question["answer_type"])
    final_correct = grade_answer(final["answer"] or None, key["gold_answer"], question["answer_type"])
    return {**question, "model": protocol["model"], "model_digest": protocol["model_digest"],
            "protocol_hash": digest, "recheck_protocol": "live-study:" + digest, "action": "recheck", "rollout": 0,
            "seed": protocol["generation"]["initial_seed"], "recheck_seed": protocol["generation"]["recheck_seed"],
            "base_call_id": initial["call_id"], "action_call_id": recheck["call_id"],
            "base_response": base["response"], "intervention_response": final["response"],
            "base_answer": base["answer"], "final_answer": final["answer"],
            "base_correct": base_correct, "final_correct": final_correct,
            "label": transition_label(base_correct, final_correct), "answer_text_changed": base["answer"] != final["answer"],
            "base_actual_tokens": base["tokens"], "action_actual_tokens": final["tokens"],
            "base_prompt_eval_count": base["prompt_eval_count"], "base_eval_count": base["eval_count"],
            "action_prompt_eval_count": final["prompt_eval_count"], "action_eval_count": final["eval_count"],
            "base_done_reason": base["done_reason"], "intervention_done_reason": final["done_reason"],
            "base_finalizer_used": False, "action_finalizer_used": False,
            "base_parse_valid": base["parse_valid"], "final_parse_valid": final["parse_valid"],
            "base_wall_seconds": base["wall_seconds"], "action_wall_seconds": final["wall_seconds"],
            "provenance": "Genuine new local Ollama initial response and unconditional offline recheck; key scored offline",
            "timestamp_utc": now()}


def collection_order(questions, smoke=0):
    # IDs are already outcome-independent seeded samples. Interleave domains and
    # partitions to avoid claiming that a time-limited prefix is a full test.
    partitions = (["source_train", "target_adaptation_pool"] if smoke else
                  ["source_train", "source_development", "target_adaptation_pool", "source_test", "target_test"])
    groups = {p: [q for q in questions if q["partition"] == p] for p in partitions}
    ordered = []
    for index in range(max(len(group) for group in groups.values())):
        ordered.extend(groups[p][index] for p in partitions if index < len(groups[p]))
    return ordered[:smoke] if smoke else ordered


def collect_study(root, *, host="http://localhost:11434", minutes=60, smoke=0, retry_failures=False,
                  resolve_uncertain=False, on_progress=None):
    root = Path(root).resolve()
    if isinstance(minutes, bool) or not isinstance(minutes, (int, float)) or not 0 < minutes <= 1440:
        raise ValueError("Collection minutes must be positive and at most 1440.")
    if isinstance(smoke, bool) or not isinstance(smoke, int) or smoke < 0:
        raise ValueError("Smoke count must be a nonnegative integer.")
    questions = read_jsonl(root / "questions.jsonl")
    keymap = validate_manifest(questions, read_jsonl(root / "trusted_keys.jsonl"))
    plan = json.loads((root / "study_plan.json").read_text(encoding="utf-8"))
    expected_hashes = plan.get("manifest_sha256", {})
    for name in ("questions.jsonl", "trusted_keys.jsonl"):
        if expected_hashes.get(name) != hashlib.sha256((root / name).read_bytes()).hexdigest():
            raise ValueError(f"Frozen study manifest hash mismatch: {name}; never silently change preselected questions or keys.")
    if smoke:
        protocol = validate_protocol(json.loads((root / "candidate_protocol.json").read_text(encoding="utf-8")))
        digest = protocol_hash(protocol)
        destination = root / "smoke"
        destination.mkdir(exist_ok=True)
        write_json(destination / "protocol.json", protocol)
        (destination / "protocol_hash.txt").write_text(digest + "\n", encoding="utf-8")
    else:
        protocol, digest = load_protocol(root)
        destination = root
    callpath, pairpath = destination / "calls.jsonl", destination / "pairs.jsonl"
    lock = destination / "collection.lock"
    session = uuid.uuid4().hex
    start = time.monotonic()
    started_at = now()
    deadline = start + minutes * 60
    status = {"status": "PARTIAL", "phase": "smoke" if smoke else "collection", "session_id": session,
              "started_at": started_at, "ended_at": None, "minutes_budget": minutes,
              "protocol_hash": digest, "errors": [], "planned_unique_questions": len(questions),
              "command": f"scripts/collect_live_study.py --study-root {root} --minutes {minutes}"}
    acquired = False
    pairs = []

    def checkpoint(reason=None):
        saved = read_jsonl(pairpath)
        journal = read_jsonl(callpath)
        counts = Counter(q["partition"] for q in saved)
        elapsed = time.monotonic() - start
        total_call_time = sum(e.get("output", {}).get("wall_seconds", 0) for e in journal if e["status"] == "succeeded")
        rate = len(saved) / total_call_time if total_call_time > 0 else None
        status.update(elapsed_seconds=elapsed, completed_pairs=len(saved), completed_unique_questions=len({p["group_id"] for p in saved}),
                      completed_by_partition=dict(counts), class_counts=dict(Counter(p["label"] for p in saved)),
                      call_events=len(journal), successful_calls=sum(e["status"] == "succeeded" for e in journal),
                      failed_calls=sum(e["status"] == "failed" for e in journal),
                      parse_failures=sum(not e["output"]["parse_valid"] for e in journal if e["status"] == "succeeded"),
                      remaining_questions=len(questions) - len(saved), measured_pairs_per_hour=rate * 3600 if rate else None,
                      estimated_remaining_hours=(len(questions) - len(saved)) / (rate * 3600) if rate else None,
                      eta_basis="sum of measured successful call wall times; includes first-call cold load; cache/backend overhead not controlled")
        if reason:
            status["stop_reason"] = reason
        write_json(destination / "collection_status.json", status)

    try:
        descriptor = os.open(lock, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
        os.write(descriptor, json.dumps({"pid": os.getpid(), "session": session, "started_at": started_at}).encode())
        os.close(descriptor)
        acquired = True
        verify_backend(protocol, host)
        events = read_jsonl(callpath)
        successes, failures, pending = journal_state(events, questions, digest, protocol)
        if pending and resolve_uncertain:
            for event in pending:
                append_jsonl(callpath, {**event, "status": "resolved_uncertain", "timestamp_utc": now(),
                                       "error": "Operator explicitly acknowledged uncertain completion; no successful response metadata exists."})
            successes, failures, pending = journal_state(read_jsonl(callpath), questions, digest, protocol)
        if pending:
            raise RuntimeError("Uncertain started call(s) cannot be silently repeated: " + ", ".join(e["call_id"] for e in pending))
        pairs = read_jsonl(pairpath)
        if len({p["example_id"] for p in pairs}) != len(pairs):
            raise ValueError("Duplicate completed pairs in saved output.")
        saved_ids = {p["example_id"] for p in pairs}
        for pair in pairs:
            if pair.get("protocol_hash") != digest:
                raise ValueError("Saved pairs do not match frozen protocol.")
            identity = pair["example_id"]
            question = next((q for q in questions if q["example_id"] == identity), None)
            if question is None or not all((identity, phase) in successes for phase in ("initial", "recheck")):
                raise ValueError("Saved pair lacks its immutable question or two actual successful calls.")
            rebuilt = build_pair(question, keymap[identity], successes[(identity, "initial")], successes[(identity, "recheck")], protocol, digest)
            if any(pair.get(key) != value for key, value in rebuilt.items() if key != "timestamp_utc"):
                raise ValueError("Saved pair differs from actual journal responses, metadata or offline trusted-key grading.")
        # Recover a pair when both calls were durably saved but pair append was interrupted.
        for q in questions:
            identity = q["example_id"]
            if identity not in saved_ids and all((identity, phase) in successes for phase in ("initial", "recheck")):
                append_jsonl(pairpath, build_pair(q, keymap[identity], successes[(identity, "initial")], successes[(identity, "recheck")], protocol, digest))
                saved_ids.add(identity)
        checkpoint()
        for q in collection_order(questions, smoke):
            identity = q["example_id"]
            if identity in saved_ids:
                continue
            for phase in ("initial", "recheck"):
                key = (identity, phase)
                if key in successes:
                    continue
                failed = [e for e in failures if (e["example_id"], e["phase"]) == key]
                if failed and not retry_failures:
                    status["status"] = "BLOCKED"
                    raise RuntimeError(f"Prior failed {phase} call for {identity}; explicit --retry-failures required.")
                if len(failed) > protocol["transport"]["explicit_retry_limit"]:
                    raise RuntimeError(f"Explicit retry limit reached for {identity}/{phase}.")
                remaining = deadline - time.monotonic()
                if remaining < protocol["transport"]["timeout_seconds"]:
                    checkpoint("insufficient session budget for one full bounded call; no request started")
                    return status
                event = {"example_id": identity, "group_id": q["group_id"], "partition": q["partition"],
                         "phase": phase, "call_id": uuid.uuid4().hex, "attempt": len(failed) + 1,
                         "protocol_hash": digest, "timestamp_utc": now(), "session_id": session,
                         "seed": protocol["generation"][phase + "_seed"], "status": "started", "smoke": bool(smoke)}
                append_jsonl(callpath, event)
                checkpoint()
                if on_progress:
                    on_progress(f"{len(saved_ids)}/{len(questions)} pairs | {identity} | {phase}")
                # A slow checkpoint/callback can consume time after the first
                # deadline check. Close the logical journal start without a
                # request rather than leave a fictitious uncertain call.
                remaining = deadline - time.monotonic()
                if remaining < protocol["transport"]["timeout_seconds"]:
                    append_jsonl(callpath, {**event, "status": "cancelled_no_request", "timestamp_utc": now(),
                                           "reason": "Session budget expired during checkpoint/progress; HTTP request never started."})
                    checkpoint("insufficient remaining session budget; request never started")
                    return status
                previous = successes[(identity, "initial")]["output"]["response"] if phase == "recheck" else None
                call_started = time.monotonic()
                try:
                    response, _, output = execute_call(protocol, q["query"], q["answer_type"], phase, previous,
                                                       host=host, timeout=min(remaining, protocol["transport"]["timeout_seconds"]))
                    completed = {**event, "status": "succeeded", "timestamp_utc": now(), "response": response, "output": output}
                    append_jsonl(callpath, completed)
                    successes[key] = completed
                    checkpoint()
                except Exception as exc:
                    failed_event = {**event, "status": "failed", "timestamp_utc": now(),
                        "error": str(exc), "error_type": type(exc).__name__, "traceback": traceback.format_exc(),
                        "wall_seconds": time.monotonic() - call_started,
                        "failure_kind": "response_metadata" if hasattr(exc, "response") else "transport_runtime",
                        "reported_response": audit_safe(getattr(exc, "response", None))}
                    append_jsonl(callpath, failed_event)
                    status["errors"].append(failed_event)
                    failures.append(failed_event)
                    status["status"] = "PARTIAL" if time.monotonic() >= deadline else "BLOCKED"
                    checkpoint("call failed; successful initial call preserved, no incorrect label manufactured")
                    return status
            pair = build_pair(q, keymap[identity], successes[(identity, "initial")], successes[(identity, "recheck")], protocol, digest)
            append_jsonl(pairpath, pair)
            saved_ids.add(identity)
            checkpoint()
            if on_progress:
                on_progress(f"Saved {identity}: {pair['label']}; actual tokens {pair['base_actual_tokens']} + {pair['action_actual_tokens']}")
        status["status"] = "COMPLETE" if not smoke and len(saved_ids) == len(questions) else "PARTIAL"
        if smoke:
            status["smoke_status"] = "COMPLETE"
        checkpoint("all selected smoke questions complete" if smoke else "all planned questions complete")
    except Exception as exc:
        status["status"] = "BLOCKED"
        status["errors"].append({"error": str(exc), "error_type": type(exc).__name__, "traceback": traceback.format_exc()})
        if acquired:
            checkpoint("validation/backend/resume blocked")
        else:
            raise
    finally:
        if acquired:
            status["ended_at"] = now()
            try:
                checkpoint()
            except Exception as exc:
                status["status"] = "BLOCKED"
                status["errors"].append({"error": f"Diagnostic checkpoint failed: {exc}",
                                          "error_type": type(exc).__name__, "traceback": traceback.format_exc()})
                write_json(destination / "collection_status.json", status)
            finally:
                lock.unlink(missing_ok=True)
    return status

def sha256_file(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def _canonical_json(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False, allow_nan=False)


def planned_counts(maths_target=1000, target_target=500):
    for name, value, minimum in (("maths_target", maths_target, 5), ("target_target", target_target, 3)):
        if isinstance(value, bool) or not isinstance(value, (int, np.integer)) or value < minimum:
            raise ValueError(f"{name} must be an integer >= {minimum} to provide nonempty reserved partitions.")
    train, development = 3 * maths_target // 5, maths_target // 5
    adaptation = 2 * target_target // 5
    return dict(source_train=int(train), source_development=int(development),
                source_test=int(maths_target - train - development),
                target_adaptation_pool=int(adaptation), target_test=int(target_target - adaptation))


def _fetch(url, retries=3):
    errors = []
    for attempt in range(1, retries + 1):
        try:
            request = Request(url, headers={"User-Agent": "RecheckLab-live-study/0.3"})
            with urlopen(request, timeout=90) as response:
                return response.read(), attempt
        except OSError as exc:
            errors.append(f"attempt {attempt}: {type(exc).__name__}: {exc}")
            if attempt < retries:
                time.sleep(attempt)
    raise RuntimeError(f"Could not download pinned benchmark resource {url}: " + "; ".join(errors))


def resolve_snapshot(repository):
    """Resolve main once; every subsequent resource URL uses the returned commit."""
    raw, _ = _fetch(f"https://huggingface.co/api/datasets/{repository}?blobs=true")
    payload = json.loads(raw)
    revision = payload.get("sha")
    if not isinstance(revision, str) or not re.fullmatch(r"[0-9a-f]{40}", revision):
        raise ValueError(f"Hugging Face did not identify a full commit for {repository}.")
    siblings = {item["rfilename"]: item for item in payload.get("siblings", []) if isinstance(item, dict) and "rfilename" in item}
    return dict(repository=repository, revision=revision, files=siblings,
                resolved_utc=datetime.now(timezone.utc).isoformat())


def download_snapshot(root, dataset, snapshot):
    config = BENCHMARKS[dataset]
    folder = Path(root) / "raw" / dataset / snapshot["revision"]
    folder.mkdir(parents=True, exist_ok=True)
    entries = {}
    for name, remote in {**config["files"], "dataset_card": "README.md"}.items():
        if remote not in snapshot["files"]:
            raise ValueError(f"Pinned {config['repository']} snapshot lacks {remote}.")
        url = f"https://huggingface.co/datasets/{config['repository']}/resolve/{snapshot['revision']}/{remote}?download=true"
        path = folder / (name + (".md" if name == "dataset_card" else ".parquet"))
        sibling = snapshot["files"][remote]
        lfs = sibling.get("lfs") or {}
        expected = lfs.get("sha256") or lfs.get("oid")
        attempts = 0
        if not path.exists():
            content, attempts = _fetch(url)
            temporary = path.with_suffix(path.suffix + ".part")
            temporary.write_bytes(content)
            actual = sha256_file(temporary)
            if expected and actual != expected:
                raise ValueError(f"Pinned download checksum mismatch for {remote}: {actual} != {expected}.")
            if name != "dataset_card":
                pd.read_parquet(temporary)  # Reject HTML/error payloads before committing cache.
            temporary.replace(path)
        actual = sha256_file(path)
        if expected and actual != expected:
            raise ValueError(f"Cached pinned benchmark checksum mismatch: {path}.")
        entries[name] = dict(path=str(path.relative_to(Path(root))), source_file=remote, url=url,
                             sha256=actual, bytes=path.stat().st_size, download_attempts=attempts)
    return entries


def _numeric_key(solution):
    if not isinstance(solution, str) or "####" not in solution:
        raise ValueError("GSM8K solution is missing its published #### answer delimiter.")
    key = solution.rsplit("####", 1)[1].strip()
    if not re.fullmatch(r"[-+]?(?:\d+|\d{1,3}(?:,\d{3})+)(?:\.\d+)?", key):
        raise ValueError(f"GSM8K published answer is not a strict numeric key: {key!r}.")
    try:
        if not Decimal(key.replace(",", "")).is_finite():
            raise ValueError("GSM8K answer key must be finite.")
    except InvalidOperation as exc:
        raise ValueError("Invalid GSM8K numeric answer key.") from exc
    return key


def benchmark_candidates(dataset, split, frame, revision):
    """Adapt benchmark schemas; gold stays private until a separate key file."""
    if dataset not in BENCHMARKS or split not in BENCHMARKS[dataset]["files"]:
        raise ValueError("Only the declared labeled benchmark splits can be selected.")
    if not re.fullmatch(r"[0-9a-f]{40}", revision):
        raise ValueError("Benchmark questions require a full pinned revision.")
    config = BENCHMARKS[dataset]
    candidates = []
    for index, source in enumerate(frame.to_dict("records")):
        stem = source.get("question")
        if not isinstance(stem, str) or not stem.strip():
            raise ValueError(f"Missing benchmark question in {dataset}/{split}/{index}.")
        if dataset == "gsm8k":
            query, identifier, key = stem.strip(), str(index), _numeric_key(source.get("answer"))
        else:
            identifier = source.get("id")
            if not isinstance(identifier, str) or not identifier.strip():
                raise ValueError(f"CommonsenseQA row {index} lacks its published question ID.")
            choices = source.get("choices")
            if isinstance(choices, dict):
                labels, texts = list(choices.get("label", [])), list(choices.get("text", []))
            elif isinstance(choices, (list, np.ndarray)):
                labels, texts = [item.get("label") for item in choices], [item.get("text") for item in choices]
            else:
                raise ValueError("CommonsenseQA choices must contain labeled answer options.")
            if len(labels) != 5 or len(texts) != 5 or set(labels) != set("ABCDE") or not all(isinstance(text, str) and text.strip() for text in texts):
                raise ValueError("CommonsenseQA requires exactly five nonempty A-E choices.")
            options = dict(zip(labels, texts))
            query = stem.strip() + "\n\nOptions:\n" + "\n".join(f"{label}. {options[label].strip()}" for label in "ABCDE")
            query += "\n\nAnswer with one option letter (A-E)."
            key = source.get("answerKey")
            if key not in tuple("ABCDE"):
                raise ValueError(f"CommonsenseQA {split}/{identifier} has no valid public answer key.")
        candidates.append(dict(example_id=f"{dataset}:{split}:{identifier}", dataset=dataset, domain=config["domain"],
            group_id=question_hash(query), query=query, answer_type=config["answer_type"],
            source_dataset=config["repository"], source_revision=revision, source_split=split,
            source_question_id=identifier, _gold_answer=key))
    if len({row["example_id"] for row in candidates}) != len(candidates):
        raise ValueError(f"Duplicate published question IDs in {dataset}/{split}.")
    return candidates


def deduplicate_candidates(pools):
    """Exclude cross-boundary overlaps globally, then choose one within-pool row."""
    locations = defaultdict(set)
    for name, rows in pools.items():
        for row in rows:
            locations[row["group_id"]].add(name)
    overlapping = {group for group, names in locations.items() if len(names) > 1}
    clean, details = {}, {}
    for name, rows in pools.items():
        kept, seen, duplicate_rows, excluded_overlap_rows = [], set(), 0, 0
        representative_keys = {}
        for row in rows:
            if row["group_id"] in overlapping:
                excluded_overlap_rows += 1
            elif row["group_id"] in seen:
                current = Decimal(row["_gold_answer"].replace(",", "")) if row["answer_type"] == "number" else row["_gold_answer"]
                if representative_keys[row["group_id"]] != current:
                    raise ValueError("Conflicting trusted keys for a canonical duplicate question.")
                duplicate_rows += 1
            else:
                kept.append(row)
                seen.add(row["group_id"])
                representative_keys[row["group_id"]] = Decimal(row["_gold_answer"].replace(",", "")) if row["answer_type"] == "number" else row["_gold_answer"]
        clean[name] = kept
        details[name] = dict(original_rows=len(rows), retained_unique_questions=len(kept),
                             removed_within_pool_duplicate_rows=duplicate_rows,
                             excluded_cross_boundary_overlap_rows=excluded_overlap_rows)
    return clean, dict(cross_boundary_overlapping_question_groups=len(overlapping), pools=details,
                      exclusion_rule="Exclude every cross-original-split/domain overlapping group; keep first pinned row for within-split duplicates before seeded selection.")


def select_questions(pools, counts, seed=42):
    clean, duplicate_audit = deduplicate_candidates(pools)
    requested = dict(gsm8k_train=counts["source_train"] + counts["source_development"], gsm8k_test=counts["source_test"],
                     commonsenseqa_train=counts["target_adaptation_pool"], commonsenseqa_validation=counts["target_test"])
    selected = {}
    seeds = {name: seed + offset for offset, name in enumerate(requested)}
    for name, number in requested.items():
        if len(clean.get(name, [])) < number:
            raise ValueError(f"{name} has too few independent questions for planned target {number} after overlap exclusion.")
        order = np.random.default_rng(seeds[name]).permutation(len(clean[name]))[:number]
        selected[name] = [dict(clean[name][index]) for index in order]
    assignments = (
        ("source_train", selected["gsm8k_train"][:counts["source_train"]]),
        ("source_development", selected["gsm8k_train"][counts["source_train"]:]),
        ("source_test", selected["gsm8k_test"]),
        ("target_adaptation_pool", selected["commonsenseqa_train"]),
        ("target_test", selected["commonsenseqa_validation"]),
    )
    questions, keys = [], []
    for partition, rows in assignments:
        for row in rows:
            key = row.pop("_gold_answer")
            row["partition"] = partition
            questions.append(row)
            keys.append(dict(example_id=row["example_id"], gold_answer=key, answer_type=row["answer_type"]))
    validate_question_manifest(questions, keys, counts)
    return questions, keys, duplicate_audit, seeds


def validate_question_manifest(questions, keys, expected_counts=None):
    if not questions:
        raise ValueError("Prepared question manifest cannot be empty.")
    identifiers, groups, counts = set(), set(), Counter()
    split_rules = {"source_train": ("gsm8k", "train"), "source_development": ("gsm8k", "train"),
                   "source_test": ("gsm8k", "test"), "target_adaptation_pool": ("commonsenseqa", "train"),
                   "target_test": ("commonsenseqa", "validation")}
    for row in questions:
        if set(row) != QUESTION_FIELDS:
            raise ValueError("Question manifest schema differs or contains an answer/future field.")
        if row["partition"] not in split_rules or (row["dataset"], row["source_split"]) != split_rules[row["partition"]]:
            raise ValueError("Question assignment crosses its reserved original benchmark split.")
        config = BENCHMARKS[row["dataset"]]
        if row["domain"] != config["domain"] or row["answer_type"] != config["answer_type"] or row["source_dataset"] != config["repository"]:
            raise ValueError("Question manifest benchmark metadata is incompatible.")
        if not re.fullmatch(r"[0-9a-f]{40}", row["source_revision"]):
            raise ValueError("Question manifest must pin a full benchmark revision.")
        if row["group_id"] != question_hash(row["query"]):
            raise ValueError("Question manifest contains a stale or forged canonical group.")
        if row["example_id"] in identifiers or row["group_id"] in groups:
            raise ValueError("Duplicate/overlapping question leaks into the fixed manifest.")
        identifiers.add(row["example_id"])
        groups.add(row["group_id"])
        counts[row["partition"]] += 1
    if expected_counts is not None and dict(counts) != expected_counts:
        raise ValueError(f"Prepared partition counts do not match the full planned targets: {dict(counts)} != {expected_counts}.")
    if len(keys) != len(questions) or {key.get("example_id") for key in keys} != identifiers:
        raise ValueError("Separate trusted keys must match every selected example exactly once.")
    types = {row["example_id"]: row["answer_type"] for row in questions}
    for key in keys:
        if set(key) != {"example_id", "gold_answer", "answer_type"} or key["answer_type"] != types[key["example_id"]]:
            raise ValueError("Trusted-key schema/type differs from its fixed question.")
        if key["answer_type"] == "number":
            _numeric_key("#### " + str(key["gold_answer"]))
        elif key["gold_answer"] not in tuple("ABCDE"):
            raise ValueError("Invalid trusted choice answer key.")
    return dict(counts)


def _read_jsonl(path):
    # JSONL boundaries are literal LF. Unicode separators inside a benchmark
    # question are valid string data and must never become record boundaries.
    return [json.loads(line) for line in Path(path).read_text(encoding="utf-8").split("\n") if line.strip()]


def _write_frozen(path, content):
    path = Path(path)
    if path.exists():
        if path.read_bytes() != content:
            raise ValueError(f"Refusing to replace a different frozen study file: {path}.")
        return
    temporary = path.with_suffix(path.suffix + ".part")
    temporary.write_bytes(content)
    temporary.replace(path)

def validate_protocol(protocol):
    if not isinstance(protocol, dict) or protocol.get("protocol_version") != PROTOCOL_VERSION:
        raise ValueError("Unsupported live-study protocol version; use its versioned implementation.")
    for key in ("model", "model_digest"):
        if not isinstance(protocol.get(key), str) or not protocol[key].strip():
            raise ValueError(f"Frozen protocol requires actual {key}.")
    if protocol.get("prompts", {}).get("initial") != INITIAL_PROMPT or protocol.get("prompts", {}).get("recheck") != RECHECK_PROMPT:
        raise ValueError("Prompt changed without a new protocol implementation/version.")
    if protocol.get("parser", {}).get("version") != PARSER_VERSION or protocol.get("grader", {}).get("version") != GRADER_VERSION:
        raise ValueError("Parser/grader changed without a new protocol version.")
    generation = protocol["generation"]
    if generation != FROZEN_GENERATION:
        raise ValueError("Generation settings changed without a new protocol implementation/version.")
    for key in ("initial_token_budget", "recheck_token_budget", "context_window"):
        value = generation.get(key)
        if isinstance(value, bool) or not isinstance(value, int) or value < 1:
            raise ValueError(f"Explicit positive integer {key} is required.")
    for key in ("initial_seed", "recheck_seed"):
        if isinstance(generation.get(key), bool) or not isinstance(generation.get(key), int):
            raise ValueError(f"Explicit integer {key} is required.")
    if not isinstance(generation.get("thinking"), bool):
        raise ValueError("Explicit thinking setting is required.")
    for key in ("temperature", "top_p", "min_p", "repeat_penalty"):
        value = generation.get(key)
        if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value) or value < 0:
            raise ValueError(f"Invalid explicit decoding setting {key}.")
    if not isinstance(generation.get("top_k"), int) or generation["top_k"] < 1 or generation.get("stream") is not False:
        raise ValueError("Explicit top_k and non-streaming calls are required.")
    if not isinstance(generation.get("stop"), list) or not all(isinstance(s, str) and s for s in generation["stop"]):
        raise ValueError("Explicit stop sequences are required.")
    if protocol.get("finalizer") != {"enabled": False, "calls": 0}:
        raise ValueError("This protocol executes no finalizer.")
    transport = protocol["transport"]
    if transport.get("automatic_retries") != 0 or transport.get("explicit_retry_limit") != 2:
        raise ValueError("This protocol uses no automatic inference retries; explicit retries are bounded.")
    if not isinstance(transport.get("timeout_seconds"), (int, float)) or not 1 <= transport["timeout_seconds"] <= 120:
        raise ValueError("Frozen call timeout must be between 1 and 120 seconds.")
    protocol_hash(protocol)  # also rejects nonfinite or non-JSON metadata
    return protocol


def load_protocol(root):
    root = Path(root)
    protocol = validate_protocol(json.loads((root / "protocol.json").read_text(encoding="utf-8")))
    expected = (root / "protocol_hash.txt").read_text(encoding="utf-8").strip()
    if protocol_hash(protocol) != expected:
        raise ValueError("Frozen protocol hash mismatch; refuse collection/inference.")
    return protocol, expected


def freeze_protocol(root, protocol):
    root = Path(root)
    root.mkdir(parents=True, exist_ok=True)
    validate_protocol(protocol)
    path = root / "protocol.json"
    if path.exists():
        existing, digest = load_protocol(root)
        if existing != protocol:
            raise ValueError("Frozen protocol differs; use a new versioned study directory.")
        return digest
    if any((root / name).exists() for name in ("calls.jsonl", "pairs.jsonl")):
        raise ValueError("Cannot freeze over existing calls without a protocol.")
    path.write_text(json.dumps(protocol, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    digest = protocol_hash(protocol)
    (root / "protocol_hash.txt").write_text(digest + "\n", encoding="utf-8")
    return digest


def verify_backend(protocol, host="http://localhost:11434"):
    actual = inspect_backend(host, protocol["model"])
    if actual["installed_model"]["digest"] != protocol["model_digest"]:
        raise ValueError("Installed checkpoint digest differs from frozen protocol.")
    frozen = protocol["backend"]
    for key in ("ollama_version", "template", "parameters", "supported_thinking"):
        if actual[key] != frozen[key]:
            raise ValueError(f"Backend {key} changed; create a new protocol version.")
    return actual


def parse_study_answer(content, answer_type):
    marked = re.findall(r"\bFINAL(?:\s+ANSWER)?\s*:\s*([^\n]+)", content, re.I)
    if not marked:
        return None
    candidate = marked[-1].strip().strip("*` ")
    if answer_type == "open":
        return candidate or None
    if answer_type == "choice":
        match = re.fullmatch(r"\(?([A-E])\)?[.!]?", candidate, re.I)
        return match[1].upper() if match else None
    if answer_type != "number":
        raise ValueError("Study answer_type must be number, choice or open.")
    return normalize_number(candidate)


def normalize_number(candidate):
    candidate = str(candidate).strip().strip("*` ")
    candidate = re.sub(r"^[$Â£â‚¹â‚¬]\s*", "", candidate).rstrip(".")
    numeric = r"[-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?"
    # Thousands separators must be groups of three; do not turn 1,5 into 15.
    if "," in candidate:
        if not re.fullmatch(r"[-+]?\d{1,3}(?:,\d{3})+(?:\.\d+)?", candidate):
            return None
        candidate = candidate.replace(",", "")
    try:
        if "/" in candidate:
            parts = candidate.split("/")
            if len(parts) != 2 or not all(re.fullmatch(numeric, p.strip()) for p in parts):
                return None
            denominator = Decimal(parts[1].strip())
            if denominator == 0:
                return None
            value = Decimal(parts[0].strip()) / denominator
        else:
            if not re.fullmatch(numeric, candidate):
                return None
            value = Decimal(candidate)
        return str(value.normalize()) if value.is_finite() else None
    except (InvalidOperation, ArithmeticError):
        return None


def grade_answer(answer, key, answer_type):
    # This function is offline only, outside the generation/feature interface.
    gold = normalize_number(key) if answer_type == "number" else str(key).strip().upper()
    if gold is None or (answer_type == "choice" and (len(gold) != 1 or gold not in "ABCDE")):
        raise ValueError("Invalid trusted benchmark key.")
    return answer is not None and answer == gold


def call_messages(protocol, question, phase, initial_response=None):
    if phase not in ("initial", "recheck") or not isinstance(question, str) or not question.strip():
        raise ValueError("A question and initial/recheck phase are required.")
    messages = [{"role": "user", "content": protocol["prompts"]["initial"] + "\n\n" + question}]
    if phase == "recheck":
        if not isinstance(initial_response, str):
            raise ValueError("Recheck requires the exact saved initial response.")
        messages += [{"role": "assistant", "content": initial_response},
                     {"role": "user", "content": protocol["prompts"]["recheck"]}]
    return messages


def normalize_call(protocol, question, answer_type, response):
    if response.get("model") != protocol["model"] or response.get("done") is not True:
        raise ValueError("Incomplete response or actual solver mismatch.")
    message = response.get("message")
    if not isinstance(message, dict) or not isinstance(message.get("content"), str):
        raise ValueError("Backend did not provide response content.")
    if "thinking" in message and not isinstance(message["thinking"], str):
        raise ValueError("Invalid actual thinking text.")
    if response.get("done_reason") not in ("stop", "length"):
        raise ValueError("Backend did not provide actual stop/length completion status.")
    tokens = actual_tokens(response)
    wall = response.get("wall_seconds")
    if isinstance(wall, bool) or not isinstance(wall, (int, float)) or not math.isfinite(wall) or wall < 0:
        raise ValueError("Measured call duration is missing or invalid.")
    parsed = parse_study_answer(message["content"], answer_type)
    text = response_text(response)
    record = validate_initial_record({"query": question, "base_response": text,
        "base_answer": parsed or "", "base_actual_tokens": tokens,
        "base_done_reason": response["done_reason"], "base_finalizer_used": False})
    output = {"response": text, "answer": parsed or "", "tokens": tokens,
              "prompt_eval_count": int(response["prompt_eval_count"]), "eval_count": int(response["eval_count"]),
              "done_reason": response["done_reason"], "wall_seconds": wall,
              "finalizer_used": False, "parse_valid": parsed is not None,
              "thinking_chars": len(message.get("thinking", "")), "model": protocol["model"]}
    return record, output


def execute_call(protocol, question, answer_type, phase, initial_response=None, *, host="http://localhost:11434", timeout=None):
    validate_protocol(protocol)
    generation = protocol["generation"]
    limit = protocol["transport"]["timeout_seconds"]
    if timeout is not None:
        limit = min(limit, timeout)
    response = ollama_chat(protocol["model"], call_messages(protocol, question, phase, initial_response), host,
                           generation[phase + "_token_budget"], generation[phase + "_seed"],
                           generation["temperature"], generation["thinking"], timeout=limit,
                           context_window=generation["context_window"],
                           decoding_options={k: generation[k] for k in ("top_k", "top_p", "min_p", "repeat_penalty", "stop")})
    try:
        record, output = normalize_call(protocol, question, answer_type, response)
    except (TypeError, ValueError, KeyError) as exc:
        exc.response = response
        raise
    return response, record, output

def _safe(value):
    """Remove keys and credentials, retaining actual token accounting fields."""
    if isinstance(value, dict):
        return {str(k): _safe(v) for k, v in value.items() if str(k).lower() not in SECRET_FIELDS}
    if isinstance(value, list):
        return [_safe(v) for v in value]
    if isinstance(value, float) and not math.isfinite(value):
        return f"{value} (invalid nonfinite value)"
    if isinstance(value, str):
        value = re.sub(r"(?i)(https?://)[^\s/@]+:[^\s/@]+@", r"\1[redacted]@", value)
        value = re.sub(r"(?i)(authorization\s*[:=]\s*(?:bearer\s+)?)[^\s\"']+", r"\1[redacted]", value)
        value = re.sub(r"(?i)((?:api[_-]?key|access[_-]?token|password)[\"']?\s*[:=]\s*[\"']?)[^\s,;\"']+", r"\1[redacted]", value)
    return value


def _json_bytes(value):
    return (json.dumps(_safe(value), indent=2, ensure_ascii=False, allow_nan=False, default=str) + "\n").encode("utf-8")


def _read_json(path, errors):
    if not path.exists():
        return {}
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
        if not isinstance(value, dict):
            raise ValueError("Expected a JSON object.")
        return value
    except (OSError, ValueError, TypeError) as exc:
        errors.append(dict(file=str(path), error=str(exc), error_type=type(exc).__name__))
        return {}


def _read_jsonl(path, errors):
    if not path.exists():
        return []
    rows = []
    try:
        lines = path.read_text(encoding="utf-8-sig").split("\n")
    except OSError as exc:
        errors.append(dict(file=str(path), error=str(exc), error_type=type(exc).__name__))
        return rows
    for line_number, line in enumerate(lines, 1):
        if not line.strip():
            continue
        try:
            row = json.loads(line)
            if not isinstance(row, dict):
                raise ValueError("Expected a JSON object.")
            rows.append(row)
        except (ValueError, TypeError) as exc:
            errors.append(dict(file=str(path), line=line_number, error=str(exc), error_type=type(exc).__name__))
    return rows


def _read_policy_rows(path, errors):
    if not path.exists():
        return []
    try:
        rows = json.loads(path.read_text(encoding="utf-8-sig"))
        if not isinstance(rows, list) or any(not isinstance(row, dict) for row in rows):
            raise ValueError("Expected a list of policy metric rows.")
        return rows
    except (OSError, ValueError, TypeError) as exc:
        errors.append(dict(file=str(path), error=str(exc), error_type=type(exc).__name__))
        return []


def _conclusion(rows, comparisons):
    frozen = next((row for row in rows if row.get("split") == "target_test" and row.get("policy") == "frozen_source_gate"), None)
    if not frozen:
        return "INCONCLUSIVE: no current target-test frozen-gate comparison is available."
    identical = comparisons.get("target_test", {}).get("completion_rule__frozen_source_gate", {}).get("identical_actions")
    if identical is True:
        return "MATCHES: frozen gate and completion rule select identical target-test actions. Accuracy and recheck costs match; gate computation adds measured overhead."
    difference, interval = frozen.get("accuracy_difference_pp_vs_completion"), frozen.get("accuracy_difference_95pct_ci_vs_completion")
    if not isinstance(difference, (int, float)) or not isinstance(interval, list) or len(interval) != 2:
        return "INCONCLUSIVE: a complete paired accuracy comparison and uncertainty are unavailable."
    if interval[0] <= 0 <= interval[1]:
        return f"INCONCLUSIVE: target-test accuracy difference is {difference:.2f} percentage points, with paired 95% interval {interval}; assess logged token, harm and latency trade-offs separately."
    finding = "higher" if interval[0] > 0 else "lower"
    return f"Frozen-gate target accuracy is {finding} than the completion rule ({difference:+.2f} percentage points; paired 95% interval {interval}). This is an accuracy finding; token, harmful-flip and reconstructed-latency costs must also be compared."


def _hash(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def _atomic_write(path, data):
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(data)
    temporary.replace(path)

def _elapsed_between(started, ended):
    try:
        start = datetime.fromisoformat(started.replace("Z", "+00:00"))
        end = datetime.fromisoformat(ended.replace("Z", "+00:00"))
        elapsed = (end - start).total_seconds()
        return elapsed if elapsed >= 0 else None
    except (AttributeError, TypeError, ValueError):
        return None


def _call_phase(row):
    phase = row.get("phase")
    return "recheck" if phase in ("revision", "recheck") else phase


def _call_identity(row):
    return row.get("call_id") or (row.get("example_id"), _call_phase(row), row.get("attempt", 1))


def _is_retry(row):
    try:
        return int(row.get("attempt", 1)) > 1
    except (TypeError, ValueError):
        return False


def _call_accounting(calls):
    succeeded = {_call_identity(row): row for row in calls if row.get("status") == "succeeded"}
    outputs = [row.get("output", row.get("response", {})) for row in succeeded.values()]
    def total(key):
        values = [row[key] for row in outputs if isinstance(row, dict) and isinstance(row.get(key), (int, float)) and math.isfinite(row[key])]
        return sum(values) if values else None
    return dict(successful_calls=len(succeeded), actual_prompt_tokens=total("prompt_eval_count"),
                actual_generated_tokens=total("eval_count"), measured_call_wall_seconds=total("wall_seconds"),
                timing_treatment="Measured individual calls include cold load where logged; backend caching is uncontrolled. Policy latency is not measured end to end here.")


def _stage(source, default_reason):
    if isinstance(source, dict) and source.get("status"):
        result = dict(source)
        reasons = result.get("not_run_reasons")
        if not reasons and result["status"] in ("NOT_RUN", "INCOMPLETE", "BLOCKED", "FAILED"):
            result["not_run_reasons"] = [result.get("reason", default_reason)]
        return result
    return {"status": "NOT_RUN", "not_run_reasons": [default_reason]}

def _bundle_entries(project_root, study_root, status, pairs, failures):
    entries = {"summaries/data_and_splits.json": _json_bytes(dict(counts=status["counts_by_partition"], hashes=status["data_hashes"], datasets=status["dataset_selection"])),
               "summaries/artifact.json": _json_bytes(status["artifact"])}
    examples, support = [], Counter()
    for row in pairs:
        label = row["label"]
        if support[label] < 2:
            examples.append(dict(kind="completed_pair", **_safe(row)))
            support[label] += 1
    for row in failures:
        category = "failure:" + str(row.get("error_type", "unknown"))
        if support[category] < 2:
            examples.append(dict(kind="failed_call", **_safe(row)))
            support[category] += 1
    entries["examples/real_records.jsonl"] = b"".join((json.dumps(row, ensure_ascii=False, default=str) + "\n").encode("utf-8") for row in examples)
    candidates = [(study_root / name, "study/" + name) for name in ("protocol.json", "protocol_hash.txt", "candidate_protocol.json", "backend_runtime_snapshot.json", "model_selection_plan.json", "smoke/protocol.json", "smoke/protocol_hash.txt", "smoke/collection_status.json", "study_plan.json", "collection_status.json", "training_status.json", "run_context.json")]
    candidates += [(path, "study/logs/" + path.name) for path in sorted((study_root / "logs").glob("*")) if path.is_file() and path.suffix in (".txt", ".log")]
    if status["stages"]["evaluation"]["status"] == "COMPLETE":
        candidates += [(path, "study/reports/" + path.name) for path in sorted((study_root / "reports").glob("*")) if path.is_file() and path.suffix in (".json", ".csv")]
    else:
        candidates += [(study_root / "reports" / name, "study/reports/" + name) for name in ("data_preparation_verified.json", "training_metadata.json", "model_selection.json", "model_configurations.json", "model_comparison.csv")]
    candidates += [(path, "source/" + str(path.relative_to(project_root)).replace("\\", "/")) for pattern in
                   ("rechecklab/study*.py", "scripts/*study*.py", "tests/test_study*.py") for path in sorted(project_root.glob(pattern))]
    candidates += [(project_root / name, "source/" + name) for name in ("app.py", "README.md", "DATA_SOURCES.md", "scripts/freeze_live_protocol.py", "rechecklab/live.py", "rechecklab/features.py", "rechecklab/predictor.py", "rechecklab/collector.py", "rechecklab/metrics.py", "tests/test_live_backend.py", "tests/test_artifact_loading.py", "requirements.txt", "requirements-core.txt")]
    budget = 8 * 1024 * 1024 - sum(len(data) for data in entries.values())
    for path, name in candidates:
        if not path.is_file() or path.stat().st_size > 1024 * 1024 or name in entries:
            continue
        data = path.read_bytes()
        if path.suffix == ".json":
            try:
                data = _json_bytes(json.loads(data.decode("utf-8-sig")))
            except (ValueError, UnicodeError):
                data = _safe(data.decode("utf-8", errors="replace")).encode("utf-8")
        else:
            data = _safe(data.decode("utf-8", errors="replace")).encode("utf-8")
        if len(data) <= budget:
            entries[name], budget = data, budget - len(data)
    artifact = Path(status["artifact"]["path"])
    if artifact.is_file() and artifact.stat().st_size < ARTIFACT_LIMIT and artifact.stat().st_size <= budget:
        entries["study/models/source_gate.joblib"] = artifact.read_bytes()
    return entries

def _json(path, value):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False, allow_nan=False) + "\n", encoding="utf-8")


def _records(path, *, allow_missing=False):
    path = Path(path)
    if allow_missing and not path.exists():
        return []
    records = []
    # JSONL is delimited by LF. Unicode line separators can occur inside a
    # valid JSON string and must not split benchmark question records.
    for line_number, line in enumerate(path.read_text(encoding="utf-8-sig").split("\n"), 1):
        if not line.strip():
            continue
        value = json.loads(line)
        if not isinstance(value, dict):
            raise ValueError(f"{path.name}:{line_number} must contain an object.")
        records.append(value)
    return records


def _digest_records(records):
    # Stable independently of collection/file ordering; excludes test records
    # when used to identify the fitted source artifact.
    ordered = sorted(records, key=lambda row: row["example_id"])
    raw = json.dumps(ordered, sort_keys=True, separators=(",", ":"), ensure_ascii=False, allow_nan=False)
    return hashlib.sha256(raw.encode("utf-8")).hexdigest()


def _counts(frame):
    return {label: int((frame.label == label).sum()) if "label" in frame else 0 for label in LABELS}


def _integer(value, name):
    if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value) or value < 0 or value % 1:
        raise ValueError(f"{name} must be a finite nonnegative integer.")
    return int(value)


def _model_configuration(value):
    """JSON representation of exact nested sklearn hyperparameters."""
    if isinstance(value, BaseEstimator):
        return {"class": type(value).__module__ + "." + type(value).__qualname__,
                "parameters": _model_configuration(value.get_params(deep=False))}
    if isinstance(value, dict):
        return {str(key): _model_configuration(item) for key, item in value.items()}
    if isinstance(value, (list, tuple)):
        return [_model_configuration(item) for item in value]
    if isinstance(value, np.ndarray):
        return _model_configuration(value.tolist())
    if isinstance(value, np.generic):
        return value.item()
    if value is None or isinstance(value, (str, bool, int, float)):
        return value
    raise ValueError(f"Cannot record estimator parameter type {type(value).__name__} faithfully.")


def _selection_plan(root, minimum, seed, budgets, seeds):
    path = root / "model_selection_plan.json"
    if not path.exists():
        return None
    declaration = json.loads(path.read_text(encoding="utf-8"))
    expected = {"minimum_training_examples_per_class": minimum, "seed": seed,
                "feature_schema": FEATURE_SCHEMA_VERSION, "feature_count": len(FEATURE_NAMES),
                "model_selection_rule": MODEL_SELECTION_RULE, "threshold_objective": PLAN_THRESHOLD_OBJECTIVE,
                "adaptation_budgets": list(budgets), "adaptation_seeds": list(seeds),
                "probabilities": "uncalibrated estimates",
                "heldout_rule": "no final evaluation until all planned held-out and adaptation pairs are collected"}
    for key, value in expected.items():
        if declaration.get(key) != value:
            raise ValueError(f"Requested training settings differ from frozen model_selection_plan: {key}.")
    return {"sha256": hashlib.sha256(path.read_bytes()).hexdigest(), "declaration": declaration}


def _load_inputs(root):
    protocol, digest = load_protocol(root)
    questions = _records(root / "questions.jsonl")
    if not questions:
        raise ValueError("The frozen question manifest is empty.")
    plan = json.loads((root / "study_plan.json").read_text(encoding="utf-8"))
    if plan.get("data_schema_version") != DATA_SCHEMA_VERSION:
        raise ValueError("Unsupported or missing frozen study data schema.")
    hash_fields = ("data_schema_version", "seed", "planned_questions_by_partition", "manifest_sha256")
    if any(key not in plan for key in hash_fields):
        raise ValueError("Frozen study plan is missing its selection/hash declaration.")
    selection_identity = {key: plan[key] for key in hash_fields}
    serialized_identity = json.dumps(selection_identity, sort_keys=True, separators=(",", ":"), ensure_ascii=False, allow_nan=False)
    if plan.get("study_plan_hash") != hashlib.sha256(serialized_identity.encode("utf-8")).hexdigest():
        raise ValueError("Frozen study plan selection hash differs from its declared contents.")
    expected = plan.get("planned_questions_by_partition")
    if not isinstance(expected, dict) or set(expected) != set(PARTITIONS):
        raise ValueError("study_plan must declare all five planned partition counts.")
    expected = {key: _integer(value, f"planned {key}") for key, value in expected.items()}
    if any(value == 0 for value in expected.values()):
        raise ValueError("All declared study partitions must contain questions.")
    seen_ids, seen_groups = set(), set()
    forbidden = {"gold_answer", "answer", "base_answer", "base_correct", "final_correct", "label"}
    for row in questions:
        for key in ("example_id", "dataset", "domain", "partition", "group_id", "query", "answer_type"):
            if not isinstance(row.get(key), str) or not row[key].strip():
                raise ValueError(f"Manifest requires explicit {key}.")
        if forbidden.intersection(row):
            raise ValueError("Question manifest must not contain answer keys or outcome labels.")
        partition = row["partition"]
        if partition not in PARTITIONS:
            raise ValueError(f"Unknown study partition: {partition}.")
        source = partition.startswith("source_")
        if (row["domain"], row["dataset"], row["answer_type"]) != (
                ("maths", "gsm8k", "number") if source else ("commonsenseqa", "commonsenseqa", "choice")):
            raise ValueError("Manifest domain/dataset/answer type disagrees with its source/target partition.")
        if row["group_id"] != question_hash(row["query"]):
            raise ValueError("Manifest group_id is not the canonical question hash.")
        if row["example_id"] in seen_ids or row["group_id"] in seen_groups:
            raise ValueError("Duplicate question or example identity across study partitions.")
        seen_ids.add(row["example_id"])
        seen_groups.add(row["group_id"])
    manifest = pd.DataFrame(questions)
    actual = {key: int((manifest.partition == key).sum()) for key in PARTITIONS}
    if actual != expected:
        raise ValueError("Frozen manifest counts differ from the declared plan; do not reduce the planned study silently.")
    keys = _records(root / "trusted_keys.jsonl")
    key_lookup = {}
    for key in keys:
        if key.get("example_id") in key_lookup or key.get("example_id") not in seen_ids:
            raise ValueError("Trusted offline keys contain duplicate or unknown question identities.")
        key_lookup[key["example_id"]] = key
    if set(key_lookup) != seen_ids:
        raise ValueError("Trusted offline keys do not cover the entire frozen question manifest.")
    if set(plan["manifest_sha256"]) != {"questions.jsonl", "trusted_keys.jsonl"}:
        raise ValueError("Frozen plan must hash BOTH the question manifest and separate trusted-key manifest.")
    for name, expected_hash in plan["manifest_sha256"].items():
        if hashlib.sha256((root / name).read_bytes()).hexdigest() != expected_hash:
            raise ValueError(f"Frozen {name} SHA-256 differs from the declared plan.")
    validate_question_manifest(questions, keys, expected)
    identity = {"model": protocol["model"], "model_digest": protocol["model_digest"],
                "action": "recheck", "recheck_protocol": "live-study:" + digest,
                "study_protocol_hash": digest, "live_study_protocol": protocol}
    raw_pairs = _records(root / "pairs.jsonl", allow_missing=True)
    calls = _records(root / "calls.jsonl", allow_missing=True)
    lookup = {row["example_id"]: row for row in questions}
    succeeded, started, terminal, successful_phases = {}, {}, set(), set()
    precision_counts = {"EXACT": 0, "UNPARSEABLE": 0, "NOT_APPLICABLE": 0}
    for event in calls:
        call_id = event.get("call_id")
        if not isinstance(call_id, str) or not call_id:
            raise ValueError("Call journal contains a missing call identity.")
        question = lookup.get(event.get("example_id"))
        if question is None or event.get("group_id") != question["group_id"] or event.get("partition") != question["partition"]:
            raise ValueError("Call journal question/group/partition differs from the frozen manifest.")
        phase = event.get("phase")
        if phase not in ("initial", "recheck") or event.get("protocol_hash") != digest or event.get("seed") != protocol["generation"][phase + "_seed"]:
            raise ValueError("Call journal phase/protocol/seed differs from the frozen generation protocol.")
        if event.get("status") == "started":
            if call_id in started:
                raise ValueError("Duplicate started call journal identity.")
            started[call_id] = event
        elif event.get("status") in ("succeeded", "failed", "resolved_uncertain", "cancelled_no_request"):
            initial_event = started.get(call_id)
            if initial_event is None or call_id in terminal:
                raise ValueError("Unmatched or duplicate terminal call journal identity.")
            for key in ("example_id", "group_id", "partition", "phase", "protocol_hash", "seed", "attempt", "session_id", "smoke"):
                if event.get(key) != initial_event.get(key):
                    raise ValueError(f"Terminal call journal event differs from its started call: {key}.")
            terminal.add(call_id)
        else:
            raise ValueError("Unknown call journal event status.")
        if event.get("status") == "succeeded":
            question_phase = (event["example_id"], event["phase"])
            if question_phase in successful_phases:
                raise ValueError("Multiple successful calls for one fixed study question/phase.")
            successful_phases.add(question_phase)
            try:
                precision = audit_numeric_precision(protocol, event.get("response"), question["answer_type"])
            except ValueError as exc:
                raise ValueError(f"Successful call {call_id} numeric precision audit failed: {exc}") from exc
            precision_counts[precision["status"]] += 1
            succeeded[call_id] = event
    for example_id, key in key_lookup.items():
        answer_type = lookup[example_id]["answer_type"]
        if key.get("answer_type") != answer_type or "gold_answer" not in key:
            raise ValueError("Trusted key type/answer is missing or differs from its frozen question.")
        if answer_type == "choice" and str(key["gold_answer"]).strip().upper() not in ("A", "B", "C", "D", "E"):
            raise ValueError("Trusted choice key must be one explicit A-E option.")
        grade_answer(None, key["gold_answer"], answer_type)
    used = set()
    for row in raw_pairs:
        example_id = row.get("example_id")
        if example_id not in lookup or example_id in used:
            raise ValueError("Pairs must contain exactly one completed pair per known question; duplicate/foreign pair found.")
        used.add(example_id)
        question = lookup[example_id]
        for key in ("group_id", "query", "dataset"):
            if row.get(key) != question[key]:
                raise ValueError(f"Pair {example_id} {key} differs from the frozen manifest.")
        for key in ("model", "model_digest", "action", "recheck_protocol"):
            if row.get(key) != identity[key]:
                raise ValueError(f"Pair {example_id} has incompatible {key}.")
        if row.get("protocol_hash") != digest:
            raise ValueError("Pair protocol hash differs from the frozen study protocol.")
        trusted = key_lookup[example_id]
        if trusted.get("answer_type") != question["answer_type"]:
            raise ValueError("Trusted key answer type differs from its frozen question.")
        # Re-validate each recorded successful backend response, then re-grade
        # its parsed answer offline. A hand-written label is not sufficient
        # evidence for training a rare class.
        for phase, prefix, answer_key, response_key, parse_key, correct_key in (
                ("initial", "base", "base_answer", "base_response", "base_parse_valid", "base_correct"),
                ("recheck", "action", "final_answer", "intervention_response", "final_parse_valid", "final_correct")):
            event = succeeded.get(row.get(prefix + "_call_id"))
            if event is None or event.get("example_id") != example_id or event.get("phase") != phase or event.get("protocol_hash") != digest:
                raise ValueError(f"Pair {example_id} lacks its matching successful {phase} call journal evidence.")
            if event.get("seed") != protocol["generation"][phase + "_seed"]:
                raise ValueError("Successful call seed differs from the frozen generation protocol.")
            _, output = normalize_call(protocol, question["query"], question["answer_type"], event.get("response", {}))
            if output != event.get("output"):
                raise ValueError("Saved normalized call output differs from its actual backend response.")
            if row.get(answer_key) != output["answer"] or row.get(response_key) != output["response"] or row.get(parse_key) is not output["parse_valid"]:
                raise ValueError("Pair answer/response/parsing metadata differs from its successful call.")
            expected_correct = grade_answer(output["answer"] or None, trusted.get("gold_answer"), question["answer_type"])
            if row.get(correct_key) is not expected_correct:
                raise ValueError("Pair correctness differs from strict offline grading against its trusted benchmark key.")
            for pair_key, output_key in ((prefix + "_actual_tokens", "tokens"), (prefix + "_prompt_eval_count", "prompt_eval_count"),
                                         (prefix + "_eval_count", "eval_count"), (prefix + "_wall_seconds", "wall_seconds")):
                if row.get(pair_key) != output[output_key]:
                    raise ValueError(f"Pair {pair_key} differs from its actual successful call.")
            done_key = "base_done_reason" if phase == "initial" else "intervention_done_reason"
            if row.get(done_key) != output["done_reason"]:
                raise ValueError("Pair completion reason differs from its actual successful call.")
        if row.get("label") != transition_label(row.get("base_correct"), row.get("final_correct")):
            raise ValueError("Recorded outcome label disagrees with the correctness transition.")
        for key in ("base_wall_seconds", "action_wall_seconds"):
            value = row.get(key)
            if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value) or value < 0:
                raise ValueError(f"Pair {example_id} requires measured finite nonnegative {key}.")
        for key in TOKEN_COMPONENTS:
            _integer(row.get(key), key)
        for stage, first, second in (("base", "base_prompt_eval_count", "base_eval_count"),
                                     ("action", "action_prompt_eval_count", "action_eval_count")):
            if _integer(row.get(stage + "_actual_tokens"), stage + "_actual_tokens") != row[first] + row[second]:
                raise ValueError(f"Pair {example_id} {stage} token total disagrees with actual call counts.")
        if row.get("base_finalizer_used") is not False:
            raise ValueError("The frozen study executes no initial finalizer.")
        if row.get("action_finalizer_used") is not False:
            raise ValueError("The frozen study executes no recheck finalizer.")
        row["partition"], row["domain"] = question["partition"], question["domain"]
    pairs = validate_pairs(pd.DataFrame(raw_pairs)) if raw_pairs else pd.DataFrame(columns=[*INITIAL_REQUIRED, "label", "group_id", "partition"])
    pairs.attrs["numeric_precision_audit"] = {"successful_call_status_counts": precision_counts,
                                             "affected_precision_loss_calls": 0,
                                             "limitation": PRECISION_LIMITATION,
                                             "scope": "All successful journal responses, including calls not yet forming a completed pair."}
    return protocol, digest, plan, manifest, pairs, identity, raw_pairs


def validate_study_inputs(root):
    """Read-only validation shared by training and the diagnostic handoff.

    Return (protocol, protocol_hash, plan, question_manifest_dataframe,
    verified_pairs_dataframe, artifact_protocol_contract, raw_pair_records).
    Empty completed pairs are valid. Invalid evidence raises ValueError; this
    function never writes files, fits a model, scores a policy or calls Ollama.
    """
    return _load_inputs(Path(root).resolve())


def _initial(frame):
    # The model never receives future responses, correctness labels or keys.
    return frame.loc[:, [*INITIAL_REQUIRED, "model", "model_digest", "action", "recheck_protocol"]].copy()


def _measured_probabilities(predictor, frame):
    probabilities, elapsed = [], []
    for index in range(len(frame)):
        start = perf_counter()
        probabilities.append(predictor.probabilities(_initial(frame.iloc[[index]]))[0])
        elapsed.append(perf_counter() - start)
    return np.asarray(probabilities), np.asarray(elapsed)


def _paired_interval(frame, decisions, completion, repeats, seed):
    final = frame.final_correct.to_numpy(dtype=float)
    base = frame.base_correct.to_numpy(dtype=float)
    difference = np.where(decisions, final, base) - np.where(completion, final, base)
    grouped = pd.DataFrame({"group": frame.group_id, "difference": difference}).groupby("group").difference.mean().to_numpy()
    rng = np.random.default_rng(seed)
    sampled = rng.choice(grouped, size=(repeats, len(grouped)), replace=True).mean(axis=1)
    return float(100 * grouped.mean()), np.quantile(100 * sampled, [.025, .975]).tolist()


def _policy_row(frame, decisions, completion, overhead, name, split, repeats, seed):
    metrics = policy_metrics(frame, decisions)
    difference, interval = _paired_interval(frame, decisions, completion, repeats, seed)
    latency = frame.base_wall_seconds.to_numpy(dtype=float) + decisions * frame.action_wall_seconds.to_numpy(dtype=float) + overhead
    return {"split": split, "policy": name, **metrics,
            "accuracy_difference_pp_vs_completion": difference,
            "accuracy_difference_95pct_ci_vs_completion": interval,
            "average_reconstructed_latency_seconds": float(latency.mean()),
            "average_predictor_overhead_seconds": float(np.mean(overhead)),
            "average_measured_initial_call_seconds": float(frame.base_wall_seconds.mean()),
            "average_measured_recheck_call_seconds": float(frame.action_wall_seconds.mean()),
            "cost_scope": COST_SCOPE,
            "latency_method": "Recorded initial wall time + selected recorded recheck wall time + measured per-question gate overhead; offline reconstruction, not deployed end-to-end timing.",
            "paired_ci_method": "Question bootstrap of paired selected-correctness difference against the completion rule."}


def _predictions(frame, probabilities, frozen, completion, overhead):
    result = frame[["example_id", "group_id", "dataset", "domain", "partition", "label", "base_correct",
                    "final_correct", "base_done_reason", "base_actual_tokens", "action_actual_tokens",
                    "base_wall_seconds", "action_wall_seconds"]].copy()
    for index, label in enumerate(LABELS):
        result["p_" + label] = probabilities[:, index]
    result["revision_score"] = revision_score(probabilities)
    result["frozen_gate_recheck"] = frozen
    result["completion_recheck"] = completion
    result["predictor_overhead_seconds"] = overhead
    return result


def _evaluate(root, parts, predictor, threshold, candidates, budgets, seeds, repeats, seed):
    reports = root / "reports"
    policies, adaptation_rows, diagnostics, comparisons, candidate_rows = [], [], {}, {}, []
    evaluated = {}
    for split in ("source_test", "target_test"):
        frame = parts[split]
        probabilities, overhead = _measured_probabilities(predictor, frame)
        frozen = decide_recheck(probabilities, threshold)
        completion = frame.base_done_reason.eq("length").to_numpy(dtype=bool)
        decisions = {"keep_initial": np.zeros(len(frame), dtype=bool), "always_recheck": np.ones(len(frame), dtype=bool),
                     "completion_rule": completion, "frozen_source_gate": frozen}
        for name, action in decisions.items():
            policies.append(_policy_row(frame, action, completion, overhead if name == "frozen_source_gate" else np.zeros(len(frame)),
                                        name, split, repeats, seed))
        diagnostics[split] = {"overall": predictor_metrics(frame, probabilities, bootstrap_repeats=repeats, seed=seed),
                              "completion_strata": {}}
        for done_reason in ("stop", "length", "unknown"):
            mask = frame.base_done_reason.eq(done_reason).to_numpy()
            if mask.any():
                stratum = frame.loc[mask].reset_index(drop=True)
                diagnostics[split]["completion_strata"][done_reason] = {
                    "predictor": predictor_metrics(stratum, probabilities[mask]),
                    "policies": [_policy_row(stratum, action[mask], completion[mask],
                                              overhead[mask] if name == "frozen_source_gate" else np.zeros(mask.sum()),
                                              name, split, repeats, seed) for name, action in decisions.items()]}
            else:
                diagnostics[split]["completion_strata"][done_reason] = {"status": "NOT_RUN", "reason": "No collected held-out questions in this completion stratum.", "questions": 0}
        comparisons[split] = {f"{first}__{second}": {"identical_actions": bool(np.array_equal(decisions[first], decisions[second])),
                                                   "different_questions": int(np.count_nonzero(decisions[first] != decisions[second]))}
                              for index, first in enumerate(decisions) for second in list(decisions)[index + 1:]}
        prediction_frame = _predictions(frame, probabilities, frozen, completion, overhead)
        evaluated[split] = (frame, probabilities, overhead, completion, frozen, prediction_frame)
        for name, candidate in candidates.items():
            candidate_rows.append({"split": split, "model": name, "selected": name == predictor.model_name,
                                   **predictor_metrics(frame, candidate.probabilities(_initial(frame)))})
    pool = parts["target_adaptation_pool"].sort_values("group_id").reset_index(drop=True)
    pool_probabilities = predictor.probabilities(_initial(pool)) if len(pool) else np.empty((0, 3))
    target, target_probabilities, overhead, completion, frozen, target_predictions = evaluated["target_test"]
    for budget in budgets:
        for subset_seed in seeds:
            if len(pool) < budget:
                adaptation_rows.append({"budget": budget, "seed": subset_seed, "status": "NOT_RUN",
                                        "reason": f"Only {len(pool)} completed target-adaptation pairs; {budget} required."})
                continue
            indices = np.random.default_rng(subset_seed).choice(len(pool), size=budget, replace=False)
            subset = pool.iloc[indices].reset_index(drop=True)
            adapted_threshold = choose_threshold(subset, revision_score(pool_probabilities[indices]))
            actions = decide_recheck(target_probabilities, adapted_threshold)
            row = _policy_row(target, actions, completion, overhead, f"adapted_budget_{budget}_seed_{subset_seed}",
                              "target_test", repeats, seed)
            row.update({"budget": budget, "seed": subset_seed, "status": "COMPLETE", "threshold": adapted_threshold,
                        "adaptation_class_counts": _counts(subset), "adaptation_example_ids": subset.example_id.tolist(),
                        "identical_to_frozen_actions": bool(np.array_equal(actions, frozen)),
                        "identical_to_completion_actions": bool(np.array_equal(actions, completion)),
                        "changed_actions_vs_frozen": int(np.count_nonzero(actions != frozen)),
                        "classifier_updated": False, "probabilities_recalibrated": False})
            row["completion_strata"] = {}
            for done_reason in ("stop", "length", "unknown"):
                mask = target.base_done_reason.eq(done_reason).to_numpy()
                row["completion_strata"][done_reason] = (
                    _policy_row(target.loc[mask].reset_index(drop=True), actions[mask], completion[mask], overhead[mask],
                                row["policy"], "target_test", repeats, seed)
                    if mask.any() else {"status": "NOT_RUN", "reason": "No held-out questions in this completion stratum.", "questions": 0})
            adaptation_rows.append(row)
            target_predictions[f"adapted_{budget}_{subset_seed}_recheck"] = actions
    adaptation_summary = []
    for budget in budgets:
        rows = [row for row in adaptation_rows if row["budget"] == budget and row["status"] == "COMPLETE"]
        summary = {"budget": budget, "completed_subsets": len(rows), "planned_subsets": len(seeds),
                   "status": "COMPLETE" if len(rows) == len(seeds) else "NOT_RUN"}
        if rows:
            for key in ("accuracy_pct", "accuracy_difference_pp_vs_completion", "harmful_flip_pct_of_all_questions",
                        "helpful_fix_pct_of_all_questions", "recheck_pct", "average_total_tokens",
                        "average_reconstructed_latency_seconds", "threshold"):
                values = [row[key] for row in rows]
                summary[key + "_mean"] = float(np.mean(values))
                summary[key + "_std_across_subsets"] = float(np.std(values, ddof=1)) if len(values) > 1 else 0.0
            summary["uncertainty"] = "Mean and sample standard deviation across adaptation subsets; each subset also has a paired question-bootstrap interval on the same target test."
        else:
            summary["reason"] = next(row["reason"] for row in adaptation_rows if row["budget"] == budget)
        adaptation_summary.append(summary)
    reports.mkdir(parents=True, exist_ok=True)
    pd.DataFrame(policies).to_csv(reports / "policy_results.csv", index=False)
    pd.DataFrame(adaptation_rows).to_csv(reports / "adaptation_runs.csv", index=False)
    pd.DataFrame(adaptation_summary).to_csv(reports / "adaptation_summary.csv", index=False)
    evaluated["source_test"][-1].to_csv(reports / "source_predictions.csv", index=False)
    target_predictions.to_csv(reports / "target_predictions.csv", index=False)
    _json(reports / "policy_results.json", policies)
    _json(reports / "adaptation_runs.json", adaptation_rows)
    _json(reports / "adaptation_summary.json", adaptation_summary)
    _json(reports / "predictor_diagnostics.json", diagnostics)
    _json(reports / "policy_action_comparison.json", comparisons)
    _json(reports / "model_comparison.json", candidate_rows)
    return adaptation_rows, adaptation_summary


def train_study(root, seed=42, minimum_per_class=MINIMUM_CLASS_SUPPORT, *,
                adaptation_budgets=ADAPTATION_BUDGETS, adaptation_seeds=ADAPTATION_SEEDS, bootstrap_repeats=600):
    """Fit only when genuine source support exists; always persist honest status.

    Missing class support is an ordinary INCOMPLETE stage, not an exception.
    Any invalid input is FAILED and no model is fitted. Neither test labels nor
    target-adaptation labels can select the classifier or frozen threshold.
    """
    root = Path(root).resolve()
    project = Path(__file__).resolve().parents[1]
    preserved = [project / name for name in ("data", "models", "reports", "outputs/v0.2")]
    if root == project or any(root == path.resolve() or path.resolve() in root.parents for path in preserved):
        raise ValueError("Use a separate versioned study directory; preserve the public pilot data/models/reports.")
    root.mkdir(parents=True, exist_ok=True)
    status = {"schema_version": 1, "updated_at_utc": datetime.now(timezone.utc).isoformat(), "status": "INCOMPLETE",
              "training": {"status": "INCOMPLETE", "not_run_reasons": []},
              "evaluation": {"status": "NOT_RUN", "not_run_reasons": []},
              "adaptation": {"status": "NOT_RUN", "not_run_reasons": []},
              "artifact": None, "probabilities": "Uncalibrated three-class classifier outputs, not correctness probabilities.",
              "limitations": [SUPPORT_LIMITATION, PRECISION_LIMITATION, COST_SCOPE],
              "predeclared": {"class_order": list(LABELS), "minimum_source_train_per_class": minimum_per_class,
                              "feature_names": list(FEATURE_NAMES), "feature_schema_version": FEATURE_SCHEMA_VERSION,
                              "feature_schema_fingerprint": FEATURE_SCHEMA_FINGERPRINT,
                              "model_selection_rule": MODEL_SELECTION_RULE, "threshold_selection_rule": THRESHOLD_RULE,
                              "harm_weight": 1.0, "token_penalty": 0.0,
                              "adaptation_budgets": list(adaptation_budgets), "adaptation_seeds": list(adaptation_seeds),
                              "source_seed": seed, "bootstrap_repeats": bootstrap_repeats}}
    try:
        if isinstance(minimum_per_class, bool) or not isinstance(minimum_per_class, int) or minimum_per_class < MINIMUM_CLASS_SUPPORT:
            raise ValueError("The predeclared study requires at least five source-training examples of EACH outcome class.")
        if isinstance(bootstrap_repeats, bool) or not isinstance(bootstrap_repeats, int) or bootstrap_repeats < 1:
            raise ValueError("Paired bootstrap requires a positive repetition count.")
        budgets = tuple(_integer(value, "adaptation budget") for value in adaptation_budgets)
        seeds = tuple(_integer(value, "adaptation seed") for value in adaptation_seeds)
        if not budgets or not seeds or any(budget < 1 for budget in budgets) or len(set(budgets)) != len(budgets) or len(set(seeds)) != len(seeds):
            raise ValueError("Adaptation budgets and seeds must be nonempty and unique; budgets must be positive.")
        selection_plan = _selection_plan(root, minimum_per_class, seed, budgets, seeds)
        status["model_selection_plan"] = selection_plan
        protocol, digest, plan, manifest, pairs, identity, raw_pairs = validate_study_inputs(root)
        planned = plan["planned_questions_by_partition"]
        completed = {name: int((pairs.partition == name).sum()) for name in PARTITIONS}
        parts = {name: pairs.loc[pairs.partition == name].reset_index(drop=True) for name in PARTITIONS}
        source_counts = _counts(parts["source_train"])
        status.update({"protocol_hash": digest, "protocol_contract": identity,
                       "study_plan_hash": plan["study_plan_hash"],
                       "planned_questions_by_partition": planned, "completed_questions_by_partition": completed,
                       "completed_pairs": len(pairs), "planned_pairs": len(manifest),
                       "source_train_class_counts": source_counts,
                       "class_counts_by_partition": {name: _counts(frame) for name, frame in parts.items()},
                       "questions_manifest_sha256": hashlib.sha256((root / "questions.jsonl").read_bytes()).hexdigest(),
                       "completed_pairs_content_sha256": _digest_records(raw_pairs)})
        status["numeric_precision_audit"] = pairs.attrs["numeric_precision_audit"]
        inadequate = {label: count for label, count in source_counts.items() if count < minimum_per_class}
        reasons = [f"Source training {label} support is {count}; at least {minimum_per_class} genuine completed pairs required."
                   for label, count in inadequate.items()]
        if parts["source_development"].empty:
            reasons.append("No completed source-development pairs are available for source-only model/threshold selection.")
        status["training"].update({"inadequate_classes": inadequate, "source_development_class_counts": _counts(parts["source_development"]),
                                    "not_run_reasons": reasons})
        if reasons:
            status["evaluation"]["not_run_reasons"] = ["Source predictor fitting has not run."]
            status["adaptation"]["not_run_reasons"] = ["No fixed source classifier/threshold exists."]
            return status
        source_records = [row for row in raw_pairs if row["partition"] in ("source_train", "source_development")]
        source_hash = _digest_records(source_records)
        predictor, selection, candidates = fit_predictor(parts["source_train"], parts["source_development"], seed=seed, return_candidates=True)
        if set(predictor.estimator.classes_) != set(LABELS):
            raise ValueError("Study estimator does not contain all three fitted outcome classes; refuse artifact.")
        dev_probabilities = predictor.probabilities(_initial(parts["source_development"]))
        threshold = choose_threshold(parts["source_development"], revision_score(dev_probabilities))
        bundle = make_artifact(predictor, threshold, protocol=identity)
        source_complete = all(completed[name] == planned[name] for name in ("source_train", "source_development"))
        configurations = {name: _model_configuration(candidate.estimator) for name, candidate in candidates.items()}
        feature_source = Path(importlib.import_module("rechecklab.features").__file__)
        bundle["study_training"] = {"status": "COMPLETE" if source_complete else "PARTIAL", "source_fit_content_sha256": source_hash,
                                    "source_train_class_counts": source_counts, "source_development_class_counts": _counts(parts["source_development"]),
                                    "source_seed": seed, "model_selection_rule": MODEL_SELECTION_RULE,
                                    "threshold_selection_rule": THRESHOLD_RULE, "target_labels_used_for_fitting": False,
                                    "test_labels_used_for_selection": False,
                                    "model_selection_plan_sha256": selection_plan["sha256"] if selection_plan else None,
                                    "model_configuration": configurations[predictor.model_name],
                                    "feature_transformations": {"implementation": "rechecklab.features.extract_features",
                                                                "source_sha256": hashlib.sha256(feature_source.read_bytes()).hexdigest(),
                                                                "inputs": list(INITIAL_REQUIRED),
                                                                "estimator_preprocessing": configurations[predictor.model_name]},
                                    "limitations": [SUPPORT_LIMITATION, PRECISION_LIMITATION, COST_SCOPE]}
        artifact_path = save_artifact(bundle, root / "models" / "source_gate.joblib")
        artifact_digest = hashlib.sha256(artifact_path.read_bytes()).hexdigest()
        restored = load_artifact(artifact_path, expected_protocol=identity, expected_sha256=artifact_digest)
        if restored["threshold"] != threshold or not np.array_equal(restored["predictor"].probabilities(_initial(parts["source_development"])), dev_probabilities):
            raise ValueError("Production artifact roundtrip changed probabilities or the frozen source threshold.")
        status["training"].update({"status": "COMPLETE" if source_complete else "PARTIAL", "model": predictor.model_name,
                                    "threshold": threshold, "source_fit_content_sha256": source_hash,
                                    "artifact_roundtrip_verified": True, "not_run_reasons": [],
                                    "limitations": [SUPPORT_LIMITATION, PRECISION_LIMITATION, COST_SCOPE,
                                                    "Held-out bootstrap intervals condition on this fitted model; they exclude training/model-selection uncertainty."] +
                                                   ["Source-development lacks " + label + "; its class-selection evidence is unavailable."
                                                    for label, count in _counts(parts["source_development"]).items() if count == 0]})
        status["artifact"] = {"path": str(artifact_path.relative_to(root)), "sha256": artifact_digest,
                              "live_activation_eligible": False,
                              "activation_reason": "Study evaluation and explicit protocol-matched activation are separate; writing an artifact does not activate live inference."}
        _json(root / "reports" / "model_selection.json", selection)
        _json(root / "reports" / "model_configurations.json", configurations)
        _json(root / "reports" / "training_metadata.json", {**bundle["study_training"], "model": predictor.model_name,
                                                            "threshold": threshold, "protocol_contract": identity,
                                                            "feature_names": list(FEATURE_NAMES), "class_order": list(LABELS),
                                                            "feature_schema_version": FEATURE_SCHEMA_VERSION,
                                                            "feature_schema_fingerprint": FEATURE_SCHEMA_FINGERPRINT,
                                                            "artifact_sha256": artifact_digest})
        required_evaluation_partitions = ("source_test", "target_test", "target_adaptation_pool") if selection_plan else ("source_test", "target_test")
        heldout_reasons = [f"{name} has {completed[name]}/{planned[name]} completed pairs; planned held-out/adaptation collection must finish before study evaluation."
                           for name in required_evaluation_partitions if completed[name] != planned[name]]
        if heldout_reasons:
            status["status"] = "PARTIAL"
            status["evaluation"]["not_run_reasons"] = heldout_reasons
            status["adaptation"]["not_run_reasons"] = ["Held-out study evaluation has not run."]
            return status
        rows, summaries = _evaluate(root, parts, predictor, threshold, candidates, budgets, seeds, bootstrap_repeats, seed)
        status["evaluation"] = {"status": "COMPLETE", "not_run_reasons": [],
                                "source_test_questions": completed["source_test"], "target_test_questions": completed["target_test"],
                                "conditional_on_partial_source_training": not source_complete,
                                "reports_directory": "reports", "selected_before_test_scoring": True}
        unavailable = [f"Budget {row['budget']}, seed {row['seed']}: {row['reason']}" for row in rows if row["status"] != "COMPLETE"]
        status["adaptation"] = {"status": "COMPLETE" if not unavailable else "PARTIAL" if any(row["status"] == "COMPLETE" for row in rows) else "NOT_RUN",
                                "not_run_reasons": unavailable, "classifier_updated": False, "probabilities_recalibrated": False,
                                "completed_runs": sum(row["status"] == "COMPLETE" for row in rows), "planned_runs": len(budgets) * len(seeds),
                                "budget_summary": summaries}
        status["status"] = "COMPLETE" if all(completed[name] == planned[name] for name in PARTITIONS) and not unavailable else "PARTIAL"
        return status
    except Exception as exc:
        status["status"] = "FAILED"
        status["error"] = f"{type(exc).__name__}: {exc}"
        status["training"]["not_run_reasons"].append(status["error"])
        if status["training"]["status"] == "INCOMPLETE":
            status["training"]["status"] = "FAILED"
        status["evaluation"] = {"status": "NOT_RUN", "not_run_reasons": [status["error"]]}
        status["adaptation"] = {"status": "NOT_RUN", "not_run_reasons": [status["error"]]}
        return status
    finally:
        _json(root / "training_status.json", status)

"""Read-only reliability checks for the already frozen live-study parser.

These checks neither reinterpret saved answers nor change the v1 grader. A
precision failure makes the record unsuitable for fitting or evaluation. A
precision-preserving parser/grader requires a future protocol version.
"""
from __future__ import annotations

from decimal import Context, Decimal, MAX_EMAX, MIN_EMIN, ROUND_HALF_EVEN, localcontext
import re

from .study_protocol import PARSER_VERSION, PROTOCOL_VERSION, parse_study_answer

AUDIT_VERSION = "frozen-decimal-precision-audit-v1"
FROZEN_DECIMAL_PRECISION = 28
PRECISION_LIMITATION = (
    "Frozen explicit-final-v1 normalization uses Decimal precision 28. "
    "Only exact finite decimal values and fractions exactly representable at "
    "that precision are accepted as reliable numeric evidence. This validation "
    "does not relabel records; precision-preserving grading needs a new version."
)


def audit_numeric_precision(protocol, response, answer_type):
    """Return audit evidence; reject numeric values rounded by the frozen parser.

    Decimal literals are compared without arithmetic or context rounding.
    Fractions are checked by exact multiplication of the parsed value and the
    original denominator. The multiplication uses sufficient coefficient
    precision and a wide exponent range; no approximate fraction expansion or
    tolerance can turn a rounded answer into reliable evidence.
    """
    parser = protocol.get("parser") if isinstance(protocol, dict) else None
    if not isinstance(protocol, dict) or protocol.get("protocol_version") != PROTOCOL_VERSION \
            or not isinstance(parser, dict) or parser.get("version") != PARSER_VERSION:
        raise ValueError("Numeric precision audit requires the frozen explicit-final-v1 study protocol.")
    if answer_type not in ("number", "choice", "open"):
        raise ValueError("Numeric precision audit requires number, choice or open answer_type.")
    evidence = {"audit_version": AUDIT_VERSION, "parser_version": PARSER_VERSION,
                "decimal_context_precision": FROZEN_DECIMAL_PRECISION,
                "applicable": answer_type == "number", "limitation": PRECISION_LIMITATION}
    if answer_type != "number":
        return {**evidence, "status": "NOT_APPLICABLE"}
    message = response.get("message") if isinstance(response, dict) else None
    if not isinstance(message, dict) or not isinstance(message.get("content"), str):
        raise ValueError("Numeric precision audit requires the actual saved response content.")
    content = message["content"]
    # Use the frozen context explicitly, without changing the caller's context.
    with localcontext(Context(prec=FROZEN_DECIMAL_PRECISION, rounding=ROUND_HALF_EVEN)):
        parsed = parse_study_answer(content, "number")
    if parsed is None:
        return {**evidence, "status": "UNPARSEABLE", "parse_valid": False,
                "exact_value_verified": False,
                "reason": "Frozen parser produced no answer; existing offline grading remains incorrect."}
    marked = re.findall(r"\bFINAL(?:\s+ANSWER)?\s*:\s*([^\n]+)", content, re.I)
    candidate = marked[-1].strip().strip("*` ")
    candidate = re.sub(r"^[$Â£â‚¹â‚¬]\s*", "", candidate).rstrip(".").replace(",", "")
    parsed_decimal = Decimal(parsed)
    if "/" in candidate:
        numerator, denominator = (Decimal(part.strip()) for part in candidate.split("/"))
        # The product's coefficient has at most the sum of coefficient lengths.
        # This avoids context-rounded division and huge exponent-expanded ints.
        coefficient_digits = len(parsed_decimal.as_tuple().digits) + len(denominator.as_tuple().digits) + 1
        with localcontext() as exact_context:
            exact_context.prec = max(FROZEN_DECIMAL_PRECISION, coefficient_digits)
            exact_context.Emax, exact_context.Emin = MAX_EMAX, MIN_EMIN
            exact_product = parsed_decimal * denominator
        if exact_product != numerator:
            raise ValueError(
                "Frozen numeric parser precision loss: fraction is nonterminating or not exactly "
                "representable at Decimal precision 28; this record cannot support reliable "
                "training/evaluation labels. A new parser/grader protocol version is required."
            )
        numeric_form = "fraction"
    else:
        exact_decimal = Decimal(candidate)
        if parsed_decimal != exact_decimal:
            raise ValueError(
                "Frozen numeric parser precision loss: Decimal precision 28 rounded the exact "
                "marked answer; this record cannot support reliable training/evaluation labels. "
                "A new parser/grader protocol version is required."
            )
        numeric_form = "decimal"
    return {**evidence, "status": "EXACT", "parse_valid": True,
            "exact_value_verified": True, "numeric_form": numeric_form}

"""RecheckLab: reproducible, lightweight self-correction experiments."""
